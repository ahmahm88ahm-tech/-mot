```typescript
/**
 * Prisma service with lifecycle hooks and soft-delete middleware.
 * Singleton — injected across all modules via PrismaModule.
 */
@Injectable()
export class PrismaService extends PrismaClient implements OnModuleInit, OnModuleDestroy {
  private readonly logger = new Logger(PrismaService.name);

  constructor() {
    super({
      log: process.env.NODE_ENV === 'development'
        ? [{ emit: 'event', level: 'query' }, { emit: 'stdout', level: 'error' }, { emit: 'stdout', level: 'warn' }]
        : [{ emit: 'stdout', level: 'error' }],
    });
  }

    await this.$connect();
    this.logger.log('Prisma connected to database');

    // Log slow queries in development (> 500ms)
    if (process.env.NODE_ENV === 'development') {
      this.$on('query' as any, (e: any) => {
        if (e.duration > 500) {
          this.logger.warn(`Slow query (${e.duration}ms): ${e.query}`);
        }
      });
    }
  }

  async onModuleDestroy() {
    await this.$disconnect();
    this.logger.log('Prisma disconnected');
  }

  /**
   * Enable graceful shutdown (NestJS hooks)
   */
  enableShutdownHooks(app: any) {
    process.on('beforeExit', () => app.close());
  }
}
```

```typescript
// apps/api/src/common/guards/jwt-auth.guard.ts
import { Injectable, ExecutionContext, UnauthorizedException } from '@nestjs/common';
import { AuthGuard } from '@nestjs/passport';
import { Reflector } from '@nestjs/core';

export const IS_PUBLIC_KEY = 'isPublic';

/**
 * JWT Authentication Guard.
 * Extends Passport's AuthGuard('jwt') and respects @Public() decorator.
 */
@Injectable()
export class JwtAuthGuard extends AuthGuard('jwt') {
  constructor(private reflector: Reflector) {
    super();
  }

  canActivate(context: ExecutionContext) {
    // Check if route is marked as public
    const isPublic = this.reflector.getAllAndOverride<boolean>(IS_PUBLIC_KEY, [
      context.getHandler(),
      context.getClass(),
    ]);
    if (isPublic) return true;

    return super.canActivate(context);
  }

  handleRequest<TUser = any>(err: any, user: TUser, info: any, context: ExecutionContext) {
    if (err || !user) {
      // Provide specific Arabic error based on JWT failure reason
      const message = this.mapJwtError(info?.name, info?.message);
      throw err || new UnauthorizedException(message);
    }
    return user;
  }

  private mapJwtError(name?: string, message?: string): string {
    switch (name) {
      case 'TokenExpiredError':
        return 'انتهت صلاحية الجلسة. يرجى تسجيل الدخول مرة أخرى.';
      case 'JsonWebTokenError':
        return 'الجلسة غير صالحة. يرجى تسجيل الدخول مرة أخرى.';
      case 'NotBeforeError':
        return 'الجلسة غير مفعّلة بعد.';
      default:
        return 'المصادقة مطلوبة للوصول لهذا المورد.';
    }
  }
}
```

```typescript
// apps/api/src/common/guards/roles.guard.ts
import { Injectable, CanActivate, ExecutionContext, ForbiddenException } from '@nestjs/common';
import { Reflector } from '@nestjs/core';
import { ROLES_KEY } from '../decorators/roles.decorator';

/**
 * Role-Based Access Control Guard.
 * Checks if authenticated user has one of the required roles.
 * Must be used AFTER JwtAuthGuard (user must be authenticated first).
 */
@Injectable()
export class RolesGuard implements CanActivate {
  constructor(private reflector: Reflector) {}

  canActivate(context: ExecutionContext): boolean {
    const requiredRoles = this.reflector.getAllAndOverride<string[]>(ROLES_KEY, [
      context.getHandler(),
      context.getClass(),
    ]);

    // No roles required = allow (auth already checked by JwtAuthGuard)
    if (!requiredRoles || requiredRoles.length === 0) return true;

    const { user } = context.switchToHttp().getRequest();
    if (!user) {
      throw new ForbiddenException('المصادقة مطلوبة');
    }

    const hasRole = requiredRoles.includes(user.role);
    if (!hasRole) {
      throw new ForbiddenException(
        `ليس لديك صلاحية تنفيذ هذا الإجراء. الصلاحيات المطلوبة: ${requiredRoles.join(' أو ')}`,
      );
    }

    return true;
  }
}
```

```typescript
// apps/api/src/common/decorators/current-user.decorator.ts
import { createParamDecorator, ExecutionContext } from '@nestjs/common';

export interface AuthUser {
  id: string;
  email: string;
  role: string;
  firstName?: string;
  lastName?: string;
}

/**
 * @CurrentUser() — extracts authenticated user from request.
 * @CurrentUser('id') — extracts specific field.
 *
 * Usage:
 *   async handler(@CurrentUser() user: AuthUser)
 *   async handler(@CurrentUser('id') userId: string)
 */
export const CurrentUser = createParamDecorator(
  (data: keyof AuthUser | undefined, ctx: ExecutionContext) => {
    const request = ctx.switchToHttp().getRequest();
    const user = request.user as AuthUser;

    if (!user) return null;
    return data ? user[data] : user;
  },
);
```

```typescript
// apps/api/src/common/filters/global-exception.filter.ts
import {
  ExceptionFilter, Catch, ArgumentsHost, HttpException, HttpStatus, Logger,
} from '@nestjs/common';
import { Response, Request } from 'express';
import { ZodError } from 'zod';
import { Prisma } from '@prisma/client';

/**
 * Global exception filter — unified error response format with Arabic messages.
 * Catches: HttpException, ZodError, Prisma errors, and unknown errors.
 * NEVER leaks internal details (stack traces, SQL, paths) to client in production.
 */
@Catch()
export class GlobalExceptionFilter implements ExceptionFilter {
  private readonly logger = new Logger(GlobalExceptionFilter.name);
  private readonly isProd = process.env.NODE_ENV === 'production';

  catch(exception: unknown, host: ArgumentsHost) {
    const ctx = host.switchToHttp();
    const response = ctx.getResponse<Response>();
    const request = ctx.getRequest<Request>();
    const requestId = (request as any).requestId || 'unknown';

    let status = HttpStatus.INTERNAL_SERVER_ERROR;
    let code = 'INTERNAL_ERROR';
    let message = 'حدث خطأ غير متوقع. يرجى المحاولة لاحقاً.';
    let details: any[] | undefined;

    if (exception instanceof ZodError) {
      // Validation error — return field-level Arabic errors
      status = HttpStatus.UNPROCESSABLE_ENTITY;
      code = 'VALIDATION_ERROR';
      message = 'البيانات المدخلة غير صالحة';
      details = exception.errors.map((e) => ({
        field: e.path.join('.'),
        message: e.message,
      }));
    } else if (exception instanceof HttpException) {
      status = exception.getStatus();
      const res = exception.getResponse();
      if (typeof res === 'string') {
        message = res;
      } else if (typeof res === 'object' && res !== null) {
        message = (res as any).message || message;
        code = (res as any).code || this.codeFromStatus(status);
        if (Array.isArray((res as any).message)) {
          details = (res as any).message.map((m: string) => ({ message: m }));
          message = 'البيانات المدخلة غير صالحة';
        }
      }
    } else if (exception instanceof Prisma.PrismaClientKnownRequestError) {
      // Map Prisma errors to safe messages
      const mapped = this.mapPrismaError(exception.code);
      status = mapped.status;
      code = mapped.code;
      message = mapped.message;
    } else if (exception instanceof Prisma.PrismaClientValidationError) {
      status = HttpStatus.BAD_REQUEST;
      code = 'DATABASE_ERROR';
      message = 'خطأ في معالجة البيانات';
    } else {
      // Unknown error — log full details internally, generic message externally
      this.logger.error({
        requestId,
        path: request.path,
        method: request.method,
        error: exception instanceof Error ? exception.stack : String(exception),
      });
    }

    // Log non-5xx at warn level for monitoring
    if (status < 500 && status >= 400) {
      this.logger.warn(`[${requestId}] ${status} ${code} ${request.method} ${request.path}: ${message}`);
    }

    response.status(status).json({
      success: false,
      error: {
        code,
        message,
        ...(details ? { details } : {}),
        requestId,
        ...(this.isProd ? {} : { timestamp: new Date().toISOString() }),
      },
    });
  }

  private codeFromStatus(status: number): string {
    const map: Record<number, string> = {
      400: 'BAD_REQUEST', 401: 'UNAUTHORIZED', 403: 'FORBIDDEN',
      404: 'NOT_FOUND', 409: 'CONFLICT', 422: 'VALIDATION_ERROR',
      429: 'RATE_LIMIT_EXCEEDED',
    };
    return map[status] || 'HTTP_ERROR';
  }

  private mapPrismaError(code: string): { status: number; code: string; message: string } {
    switch (code) {
      case 'P2002': // Unique constraint failed
        return { status: 409, code: 'CONFLICT', message: 'هذا السجل موجود مسبقاً (قيمة مكررة).' };
      case 'P2025': // Record not found
        return { status: 404, code: 'NOT_FOUND', message: 'السجل المطلوب غير موجود.' };
      case 'P2003': // Foreign key constraint
        return { status: 400, code: 'REFERENCE_ERROR', message: 'لا يمكن تنفيذ العملية لوجود بيانات مرتبطة.' };
      case 'P2014': // Relation violation
        return { status: 400, code: 'RELATION_ERROR', message: 'خطأ في العلاقة بين البيانات.' };
      default:
        return { status: 500, code: 'DATABASE_ERROR', message: 'خطأ في قاعدة البيانات. يرجى المحاولة لاحقاً.' };
    }
  }
}
```

```typescript
// apps/api/src/modules/auth/password.service.ts
import { Injectable, BadRequestException } from '@nestjs/common';
import * as bcrypt from 'bcrypt';

/**
 * Password hashing + policy validation.
 * Matches 09-Security §4.1 password policy exactly.
 */
@Injectable()
export class PasswordService {
  private readonly SALT_ROUNDS = 12; // bcrypt cost factor

  // Common compromised passwords (subset — in production use HaveIBeenPwned API)
  private readonly COMMON_PASSWORDS = new Set([
    'password', '123456', '12345678', 'qwerty', 'abc123',
    'password123', 'admin123', 'letmein', 'welcome', 'monkey',
  ]);

  /**
   * Hash password with bcrypt
   */
  async hash(password: string): Promise<string> {
    return bcrypt.hash(password, this.SALT_ROUNDS);
  }

  /**
   * Verify password against hash
   */
  async verify(password: string, hash: string): Promise<boolean> {
    return bcrypt.compare(password, hash);
  }

  /**
   * Validate password against policy. Throws BadRequestException with Arabic message.
   */
  validatePolicy(password: string, userInfo?: { firstName?: string; lastName?: string; email?: string }): void {
    const errors: string[] = [];

    if (password.length < 10) errors.push('10 أحرف على الأقل');
    if (password.length > 128) errors.push('128 حرف كحد أقصى');
    if (!/[A-Z]/.test(password) && !/[A-Z]/.test(password)) {
      // For Arabic-context users, require either uppercase latin OR a digit+special combo
      if (!/[A-Z]/.test(password)) errors.push('حرف لاتيني كبير واحد (A-Z)');
    }
    if (!/[a-z]/.test(password)) errors.push('حرف لاتيني صغير واحد (a-z)');
    if (!/[0-9]/.test(password)) errors.push('رقم واحد (0-9)');
    if (!/[^A-Za-z0-9]/.test(password)) errors.push('رمز خاص واحد (!@#$%...)');
    if (this.COMMON_PASSWORDS.has(password.toLowerCase())) errors.push('كلمة المرور شائعة جداً وسهلة الاختراق');

    // Disallow user info in password
    if (userInfo) {
      const lower = password.toLowerCase();
      if (userInfo.email && lower.includes(userInfo.email.split('@')[0].toLowerCase())) {
        errors.push('لا يجب أن تحتوي كلمة المرور على بريدك الإلكتروني');
      }
      if (userInfo.firstName && lower.includes(userInfo.firstName.toLowerCase())) {
        errors.push('لا يجب أن تحتوي كلمة المرور على اسمك');
      }
    }

    // Disallow sequential patterns
    if (/(012|123|234|345|456|567|678|789|abc|bcd|cde|def)/i.test(password)) {
      errors.push('لا يجب أن تحتوي على تسلسلات متتالية (123, abc)');
    }

    if (errors.length > 0) {
      throw new BadRequestException({
        code: 'WEAK_PASSWORD',
        message: 'كلمة المرور لا تستوفي متطلبات الأمان',
        requirements: errors,
      });
    }
  }
}
```

```typescript
// apps/api/src/modules/auth/jwt.strategy.ts
import { Injectable, UnauthorizedException } from '@nestjs/common';
import { PassportStrategy } from '@nestjs/passport';
import { ExtractJwt, Strategy } from 'passport-jwt';
import { PrismaService } from '../../prisma/prisma.service';

export interface JwtPayload {
  sub: string;       // user id
  email: string;
  role: string;
  type: 'access';    // distinguish from refresh
  iat?: number;
  exp?: number;
}

/**
 * Passport JWT strategy — validates access tokens on protected routes.
 * Checks token validity AND user active status (revoked/suspended users rejected).
 */
@Injectable()
export class JwtStrategy extends PassportStrategy(Strategy, 'jwt') {
  constructor(private readonly prisma: PrismaService) {
    super({
      jwtFromRequest: ExtractJwt.fromAuthHeaderAsBearerToken(),
      ignoreExpiration: false,
      secretOrKey: process.env.JWT_SECRET || 'dev-secret-change-me-min-32-chars!!',
      // In production with RS256, use:
      // secretOrKeyProvider: ... or pass public key
    });
  }

  async validate(payload: JwtPayload) {
    // Reject non-access tokens
    if (payload.type !== 'access') {
      throw new UnauthorizedException('نوع الرمز غير صالح');
    }

    // Verify user still exists and is active
    const user = await this.prisma.user.findUnique({
      where: { id: payload.sub },
      select: { id: true, email: true, role: true, status: true, firstName: true, lastName: true },
    });

    if (!user) throw new UnauthorizedException('المستخدم غير موجود');
    if (user.status !== 'ACTIVE') {
      const msg = user.status === 'SUSPENDED'
        ? 'حسابك موقوف. تواصل مع الدعم.'
        : 'حسابك غير نشط.';
      throw new UnauthorizedException(msg);
    }

    return user; // attached to request.user
  }
}
```

```typescript
// apps/api/src/modules/auth/mfa.service.ts
import { Injectable, BadRequestException } from '@nestjs/common';
import { authenticator } from 'otplib';
import * as QRCode from 'qrcode';
import * as crypto from 'crypto';
import { PrismaService } from '../../prisma/prisma.service';

@Injectable()
export class MfaService {
  constructor(private readonly prisma: PrismaService) {
    authenticator.options = { window: 1, digits: 6 }; // ±1 step tolerance
  }

  /**
   * Generate MFA secret + QR code for setup
   */
  async generateSetup(userId: string, email: string) {
    const secret = authenticator.generateSecret();
    const otpAuthUrl = authenticator.keyuri(email, 'MOTANAQIL', secret);
    const qrCodeDataUrl = await QRCode.toDataURL(otpAuthUrl);

    // Store encrypted secret temporarily (enabled=false until verified)
    const encrypted = this.encrypt(secret);
    await this.prisma.userMFA.upsert({
      where: { userId },
      create: { userId, secretEncrypted: encrypted, enabled: false },
      update: { secretEncrypted: encrypted, enabled: false },
    });

    return { secret, qrCode: qrCodeDataUrl, otpAuthUrl };
  }

  /**
   * Verify token and enable MFA; generate backup codes
   */
  async enable(userId: string, token: string) {
    const mfa = await this.prisma.userMFA.findUnique({ where: { userId } });
    if (!mfa) throw new BadRequestException('لم يتم إعداد MFA. ابدأ بالإعداد أولاً.');

    const secret = this.decrypt(mfa.secretEncrypted);
    if (!authenticator.verify({ token, secret })) {
      throw new BadRequestException('رمز التحقق غير صحيح. تأكد من التطبيق وحاول مجدداً.');
    }

    // Generate 10 backup codes (hashed)
    const backupCodes = Array.from({ length: 10 }, () =>
      crypto.randomBytes(4).toString('hex').match(/.{1,4}/g)!.join('-'),
    );
    const hashedCodes = backupCodes.map((c) => this.hashSimple(c));

    await this.prisma.userMFA.update({
      where: { userId },
      data: { enabled: true, backupCodes: hashedCodes, enabledAt: new Date() },
    });

    return { backupCodes }; // show ONCE to user
  }

  /**
   * Verify MFA token (login step 2) or backup code
   */
  async verify(userId: string, token: string): Promise<boolean> {
    const mfa = await this.prisma.userMFA.findUnique({ where: { userId } });
    if (!mfa || !mfa.enabled) return true; // MFA not enabled = pass

    const secret = this.decrypt(mfa.secretEncrypted);

    // Try TOTP
    if (authenticator.verify({ token, secret })) return true;

    // Try backup code
    const hashed = this.hashSimple(token);
    if (mfa.backupCodes.includes(hashed)) {
      // Consume backup code (one-time use)
      await this.prisma.userMFA.update({
        where: { userId },
        data: { backupCodes: mfa.backupCodes.filter((c) => c !== hashed) },
      });
      return true;
    }

    return false;
  }

  async disable(userId: string) {
    await this.prisma.userMFA.update({
      where: { userId },
      data: { enabled: false, secretEncrypted: '', backupCodes: [] },
    });
  }

  async isEnabled(userId: string): Promise<boolean> {
    const mfa = await this.prisma.userMFA.findUnique({ where: { userId } });
    return mfa?.enabled ?? false;
  }

  // ===== Crypto helpers (in production use KMS) =====
  private encrypt(plaintext: string): string {
    const key = this.getKey();
    const iv = crypto.randomBytes(16);
    const cipher = crypto.createCipheriv('aes-256-gcm', key, iv);
    const enc = Buffer.concat([cipher.update(plaintext, 'utf8'), cipher.final()]);
    const tag = cipher.getAuthTag();
    return `${iv.toString('hex')}:${tag.toString('hex')}:${enc.toString('hex')}`;
  }

  private decrypt(payload: string): string {
    const [ivHex, tagHex, encHex] = payload.split(':');
    const key = this.getKey();
    const decipher = crypto.createDecipheriv('aes-256-gcm', key, Buffer.from(ivHex, 'hex'));
    decipher.setAuthTag(Buffer.from(tagHex, 'hex'));
    return decipher.update(Buffer.from(encHex, 'hex')) + decipher.final('utf8');
  }

  private getKey(): Buffer {
    const secret = process.env.MFA_ENCRYPTION_KEY || process.env.JWT_SECRET || 'fallback-key-32chars-minimum!!!!';
    return crypto.createHash('sha256').update(secret).digest();
  }

  private hashSimple(value: string): string {
    return crypto.createHash('sha256').update(value).digest('hex');
  }
}
```

```typescript
// apps/api/src/modules/auth/auth.service.ts
import {
  Injectable, UnauthorizedException, ConflictException, BadRequestException,
  NotFoundException, Logger,
} from '@nestjs/common';
import { JwtService } from '@nestjs/jwt';
import { EventEmitter2 } from '@nestjs/event-emitter';
import * as crypto from 'crypto';
import { PrismaService } from '../../prisma/prisma.service';
import { PasswordService } from './password.service';
import { MfaService } from './mfa.service';
import type { RegisterDto, LoginDto, RefreshDto, ForgotPasswordDto, ResetPasswordDto, ChangePasswordDto } from './dto/auth.dto';

@Injectable()
export class AuthService {
  private readonly logger = new Logger(AuthService.name);
  private readonly ACCESS_TOKEN_EXPIRY = '15m';
  private readonly REFRESH_TOKEN_EXPIRY_MS = 7 * 24 * 60 * 60 * 1000; // 7 days
  private readonly RESET_TOKEN_EXPIRY_MS = 60 * 60 * 1000; // 1 hour

  constructor(
    private readonly prisma: PrismaService,
    private readonly jwtService: JwtService,
    private readonly passwordService: PasswordService,
    private readonly mfaService: MfaService,
    private readonly eventEmitter: EventEmitter2,
  ) {}

  // ===== REGISTER =====
  async register(dto: RegisterDto, ip?: string, userAgent?: string) {
    // Check uniqueness
    const existing = await this.prisma.user.findFirst({
      where: { OR: [{ email: dto.email }, { phone: dto.phone }] },
    });
    if (existing) {
      const field = existing.email === dto.email ? 'البريد الإلكتروني' : 'رقم الجوال';
      throw new ConflictException(`${field} مسجل مسبقاً`);
    }

    // Validate password policy
    this.passwordService.validatePolicy(dto.password, {
      firstName: dto.firstName, lastName: dto.lastName, email: dto.email,
    });

    const passwordHash = await this.passwordService.hash(dto.password);

    const user = await this.prisma.user.create({
      data: {
        firstName: dto.firstName,
        lastName: dto.lastName,
        email: dto.email,
        phone: dto.phone,
        passwordHash,
        cityId: dto.cityId ?? null,
        role: 'CUSTOMER',
      },
    });

    const tokens = await this.generateTokens(user.id, user.email, user.role);

    this.logger.log(`User registered: ${user.email} (${user.id})`);
    this.eventEmitter.emit('auth.registered', { userId: user.id, email: user.email });

    return { user: this.sanitizeUser(user), ...tokens };
  }

  // ===== LOGIN =====
  async login(dto: LoginDto, ip?: string, userAgent?: string) {
    // Find by email or phone
    const user = await this.prisma.user.findFirst({
      where: {
        OR: [
          { email: dto.identifier.toLowerCase().trim() },
          { phone: this.normalizePhone(dto.identifier) },
        ],
      },
    });

    if (!user || !(await this.passwordService.verify(dto.password, user.passwordHash))) {
      // Generic message to prevent enumeration
      throw new UnauthorizedException('البريد/الجوال أو كلمة المرور غير صحيحة');
    }

    if (user.status !== 'ACTIVE') {
      throw new UnauthorizedException(
        user.status === 'SUSPENDED' ? 'حسابك موقوف. تواصل مع الدعم.' : 'حسابك غير نشط.',
      );
    }

    // Check MFA
    const mfaEnabled = await this.mfaService.isEnabled(user.id);
    if (mfaEnabled) {
      // Issue temporary token for MFA verification step
      const tempToken = this.jwtService.sign(
        { sub: user.id, type: 'mfa_pending' },
        { expiresIn: '5m' },
      );
      return { requiresMfa: true, tempToken };
    }

    // Update last login + issue tokens
    await this.prisma.user.update({
      where: { id: user.id },
      data: { lastLoginAt: new Date(), lastLoginIp: ip ?? null },
    });

    const tokens = await this.generateTokens(user.id, user.email, user.role);

    this.eventEmitter.emit('auth.login', { userId: user.id, ip, userAgent });

    return { user: this.sanitizeUser(user), ...tokens, requiresMfa: false };
  }

  // ===== MFA VERIFY (login step 2) =====
  async verifyMfa(tempToken: string, mfaToken: string) {
    let payload: any;
    try {
      payload = this.jwtService.verify(tempToken);
    } catch {
      throw new UnauthorizedException('انتهت صلاحية رمز التحقق المؤقت. ابدأ تسجيل الدخول مجدداً.');
    }
    if (payload.type !== 'mfa_pending') {
      throw new UnauthorizedException('رمز مؤقت غير صالح');
    }

    const valid = await this.mfaService.verify(payload.sub, mfaToken);
    if (!valid) throw new UnauthorizedException('رمز التحقق غير صحيح');

    const user = await this.prisma.user.findUnique({ where: { id: payload.sub } });
    if (!user || user.status !== 'ACTIVE') throw new UnauthorizedException('المستخدم غير صالح');

    await this.prisma.user.update({ where: { id: user.id }, data: { lastLoginAt: new Date() } });
    const tokens = await this.generateTokens(user.id, user.email, user.role);

    return { user: this.sanitizeUser(user), ...tokens };
  }

  // ===== REFRESH =====
  async refresh(dto: RefreshDto) {
    const tokenHash = this.hashToken(dto.refreshToken);
    const stored = await this.prisma.refreshToken.findUnique({ where: { tokenHash } });

    if (!stored) throw new UnauthorizedException('رمز التحديث غير صالح');

    // Reuse detection — if already revoked, attacker may have stolen token
    if (stored.revokedAt) {
      // Revoke entire family (all tokens for this user)
      await this.prisma.refreshToken.updateMany({
        where: { userId: stored.userId, revokedAt: null },
        data: { revokedAt: new Date() },
      });
      this.logger.warn(`Refresh token reuse detected for user ${stored.userId} — all sessions revoked`);
      this.eventEmitter.emit('auth.token_reuse', { userId: stored.userId });
      throw new UnauthorizedException('تم اكتشاف استخدام غير آمن. تم تسجيل خروجك من جميع الأجهزة.');
    }

    if (stored.expiresAt < new Date()) {
      throw new UnauthorizedException('انتهت صلاحية رمز التحديث. سجل دخولك مجدداً.');
    }

    // Revoke current token (rotation)
    await this.prisma.refreshToken.update({
      where: { id: stored.id },
      data: { revokedAt: new Date() },
    });

    const user = await this.prisma.user.findUnique({ where: { id: stored.userId } });
    if (!user || user.status !== 'ACTIVE') throw new UnauthorizedException('المستخدم غير صالح');

    return this.generateTokens(user.id, user.email, user.role);
  }

  // ===== LOGOUT =====
  async logout(userId: string, refreshToken?: string) {
    if (refreshToken) {
      await this.prisma.refreshToken.updateMany({
        where: { tokenHash: this.hashToken(refreshToken), userId },
        data: { revokedAt: new Date() },
      });
    }
    return { success: true };
  }

  async logoutAll(userId: string) {
    await this.prisma.refreshToken.updateMany({
      where: { userId, revokedAt: null },
      data: { revokedAt: new Date() },
    });
    return { success: true };
  }

  // ===== FORGOT PASSWORD =====
  async forgotPassword(dto: ForgotPasswordDto) {
    const user = await this.prisma.user.findUnique({ where: { email: dto.email } });

    // Always return success to prevent enumeration
    if (!user) return { success: true, message: 'إذا كان البريد مسجلاً، ستصلك رسالة الاستعادة.' };

    const token = crypto.randomBytes(32).toString('hex');
    const tokenHash = this.hashToken(token);

    await this.prisma.passwordResetToken.create({
      data: {
        userId: user.id,
        tokenHash,
        expiresAt: new Date(Date.now() + this.RESET_TOKEN_EXPIRY_MS),
      },
    });

    // In production: queue email with reset link containing `token`
    this.eventEmitter.emit('auth.password_reset_requested', { userId: user.id, email: user.email, token });

    return { success: true, message: 'إذا كان البريد مسجلاً، ستصلك رسالة الاستعادة.' };
  }

  // ===== RESET PASSWORD =====
  async resetPassword(dto: ResetPasswordDto) {
    const tokenHash = this.hashToken(dto.token);
    const stored = await this.prisma.passwordResetToken.findUnique({ where: { tokenHash } });

    if (!stored || stored.usedAt || stored.expiresAt < new Date()) {
      throw new BadRequestException('رابط الاستعادة غير صالح أو منتهي الصلاحية');
    }

    const user = await this.prisma.user.findUnique({ where: { id: stored.userId } });
    if (!user) throw new NotFoundException('المستخدم غير موجود');

    this.passwordService.validatePolicy(dto.newPassword, {
      firstName: user.firstName, lastName: user.lastName, email: user.email,
    });

    const passwordHash = await this.passwordService.hash(dto.newPassword);

    await this.prisma.$transaction([
      this.prisma.user.update({ where: { id: user.id }, data: { passwordHash } }),
      this.prisma.passwordResetToken.update({ where: { id: stored.id }, data: { usedAt: new Date() } }),
      // Revoke all refresh tokens (force re-login)
      this.prisma.refreshToken.updateMany({ where: { userId: user.id, revokedAt: null }, data: { revokedAt: new Date() } }),
    ]);

    this.eventEmitter.emit('auth.password_changed', { userId: user.id });

    return { success: true };
  }

  // ===== CHANGE PASSWORD (authenticated) =====
  async changePassword(userId: string, dto: ChangePasswordDto) {
    const user = await this.prisma.user.findUnique({ where: { id: userId } });
    if (!user) throw new NotFoundException('المستخدم غير موجود');

    if (!(await this.passwordService.verify(dto.currentPassword, user.passwordHash))) {
      throw new UnauthorizedException('كلمة المرور الحالية غير صحيحة');
    }

    this.passwordService.validatePolicy(dto.newPassword, {
      firstName: user.firstName, lastName: user.lastName, email: user.email,
    });

    const passwordHash = await this.passwordService.hash(dto.newPassword);
    await this.prisma.$transaction([
      this.prisma.user.update({ where: { id: userId }, data: { passwordHash } }),
      this.prisma.refreshToken.updateMany({ where: { userId, revokedAt: null }, data: { revokedAt: new Date() } }),
    ]);

    this.eventEmitter.emit('auth.password_changed', { userId });
    return { success: true };
  }

  // ===== ME =====
  async getMe(userId: string) {
    const user = await this.prisma.user.findUnique({
      where: { id: userId },
      include: { city: { select: { id: true, nameAr: true, slug: true } } },
    });
    if (!user) throw new NotFoundException('المستخدم غير موجود');
    const mfaEnabled = await this.mfaService.isEnabled(userId);
    return { ...this.sanitizeUser(user), mfaEnabled };
  }

  // ===== MFA SETUP =====
  async setupMfa(userId: string) {
    const user = await this.prisma.user.findUnique({ where: { id: userId } });
    if (!user) throw new NotFoundException('المستخدم غير موجود');
    return this.mfaService.generateSetup(userId, user.email);
  }

  async enableMfa(userId: string, token: string) {
    return this.mfaService.enable(userId, token);
  }

  async disableMfa(userId: string) {
    await this.mfaService.disable(userId);
    return { success: true };
  }

  // ===== PRIVATE HELPERS =====
  private async generateTokens(userId: string, email: string, role: string) {
    const payload = { sub: userId, email, role, type: 'access' as const };
    const accessToken = this.jwtService.sign(payload, { expiresIn: this.ACCESS_TOKEN_EXPIRY });

    // Refresh token — opaque, stored hashed
    const refreshToken = crypto.randomBytes(48).toString('hex');
    const tokenHash = this.hashToken(refreshToken);
    const familyId = crypto.randomUUID();

    await this.prisma.refreshToken.create({
      data: {
        userId,
        tokenHash,
        familyId,
        expiresAt: new Date(Date.now() + this.REFRESH_TOKEN_EXPIRY_MS),
      },
    });

    return { accessToken, refreshToken, expiresIn: 900 }; // 15 min in seconds
  }

  private hashToken(token: string): string {
    return crypto.createHash('sha256').update(token).digest('hex');
  }

  private normalizePhone(input: string): string {
    const cleaned = input.replace(/\s/g, '');
    if (cleaned.startsWith('+966')) return cleaned;
    if (cleaned.startsWith('0')) return `+966${cleaned.slice(1)}`;
    return `+966${cleaned}`;
  }

  private sanitizeUser(user: any) {
    return {
      id: user.id,
      firstName: user.firstName,
      lastName: user.lastName,
      email: user.email,
      phone: user.phone,
      role: user.role,
      avatarUrl: user.avatarUrl ?? null,
      emailVerified: user.emailVerified ?? false,
      phoneVerified: user.phoneVerified ?? false,
      city: user.city ?? null,
    };
  }
```

### ⚠️ ملاحظة مهمة عن Prisma Schema

الـ Auth Module يستخدم جداول **لم تُضف بعد** للـ schema الذي كتبناه في المرحلة 2/5:
- `refresh_tokens` (مع `familyId`)
- `password_reset_tokens`
- `user_mfa`

يجب إضافة هذه للـ `schema.prisma`. إليك الإضافة المطلوبة:

```prisma
// أضف لـ schema.prisma

model RefreshToken {
  id         String    @id @default(cuid())
  userId     String    @map("user_id")
  user       User      @relation(fields: [userId], references: [id], onDelete: Cascade)
  tokenHash  String    @unique @map("token_hash")
  familyId   String    @map("family_id")
  expiresAt  DateTime  @map("expires_at")
  revokedAt  DateTime? @map("revoked_at")
  createdAt  DateTime  @default(now()) @map("created_at")

  @@index([userId])
  @@index([familyId])
  @@index([expiresAt])
  @@map("refresh_tokens")
}

model PasswordResetToken {
  id        String    @id @default(cuid())
  userId    String    @map("user_id")
  user      User      @relation(fields: [userId], references: [id], onDelete: Cascade)
  tokenHash String    @unique @map("token_hash")
  expiresAt DateTime  @map("expires_at")
  usedAt    DateTime? @map("used_at")
  createdAt DateTime  @default(now()) @map("created_at")

  @@index([userId])
  @@map("password_reset_tokens")
}

model UserMFA {
  id              String    @id @default(cuid())
  userId          String    @unique @map("user_id")
  user            User      @relation(fields: [userId], references: [id], onDelete: Cascade)
  secretEncrypted String    @default("") @map("secret_encrypted") @db.Text
  enabled         Boolean   @default(false)
  enabledAt       DateTime? @map("enabled_at")
  backupCodes     String[]  @default([]) @map("backup_codes")
  updatedAt       DateTime  @updatedAt @map("updated_at")

  @@map("user_mfa")
}
```

وأضف العلاقات لـ model `User`:
```prisma
model User {
  // ... الحقول الموجودة ...
  refreshTokens      RefreshToken[]
  passwordResetTokens PasswordResetToken[]
  userMFA            UserMFA?
}
```
