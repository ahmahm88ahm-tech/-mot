# 🚀 التنفيذ الفعلي: ربط الـ Frontend بالـ API الحقيقية (End-to-End)

> قبل الكود، **شفافية كاملة** عن الفجوات الحقيقية التي وجدتها بمراجعة الكود المرجعي السابق (المراحل 1-3) — لأن الربط الحقيقي يعني سدها جميعاً، لا كتابة fetch جديد فوق القديم:

| # | الفجوة الحقيقية | أثرها |
|---|----------------|-------|
| F1 | لا `apiClient` موحد — كل component يكتب `fetch` يدوياً | لا refresh token، لا معالجة أخطاء موحدة، تكرار |
| F2 | `request/page.tsx` يستخدم `credentials:'include'` فقط | الـ Backend يقرأ **Bearer header** لا cookie → **401 على كل طلب** |
| F3 | لا endpoint `/tracking/order/:orderNumber` في الـ Backend | صفحة التتبع **معطلة تماماً** |
| F4 | `auth-store` لا يتعامل مع `requiresMfa` | تسجيل دخول MFA **مكسور** |
| F5 | `dashboard` بياناته hardcoded (أصفار) | لا يعكس DB |
| F6 | الـ steps تكتب fetch بدون base URL/token | 401 + CORS |
| F7 | أخطاء Zod العربية لا تظهر في الـ UI | المستخدم يرى "خطأ غير متوقع" بدل الرسالة |
| F8 | `logout` لا يمرر `refreshToken` | الـ token يبقى صالحاً بعد الخروج |

كل ملف أدناه يسد فجوة محددة (مشار إليها). النسخة هنا **تحل محل** النسخ المرجعية السابقة لنفس الملفات.

```typescript
// apps/api/src/modules/orders/tracking.controller.ts
// endpoint عام (بدون auth) للتتبع برقم الطلب — يرجع بيانات آمنة فقط.
import { Controller, Get, Param, NotFoundException } from '@nestjs/common';
import { ApiTags, ApiOperation } from '@nestjs/swagger';
import { PrismaService } from '../../prisma/prisma.service';
import { ORDER_STATUS_LABELS } from './constants';

@ApiTags('Tracking (Public)')
@Controller({ path: 'tracking', version: '1' })
export class TrackingController {
  constructor(private readonly prisma: PrismaService) {}

  @Get('order/:orderNumber')
  @ApiOperation({ summary: 'تتبع طلب برقمه (عام)' })
  async track(@Param('orderNumber') orderNumber: string) {
    const order = await this.prisma.order.findUnique({
      where: { orderNumber, deletedAt: null },
      select: {
        id: true, orderNumber: true, status: true,
        scheduledDate: true, scheduledSlot: true,
        fromLatitude: true, fromLongitude: true, fromAddress: true,
        toLatitude: true, toLongitude: true, toAddress: true,
        createdAt: true,
        // بيانات السائق الآمنة فقط (لا email/لا id داخلي حساس)
        assignedTo: { select: { firstName: true, lastName: true, phone: true, avatarUrl: true } },
        service: { select: { nameAr: true } },
        fromCity: { select: { nameAr: true } },
        toCity: { select: { nameAr: true } },
      },
    });
    if (!order) throw new NotFoundException('الطلب غير موجود');

    // آخر موقع للسائق من سجل الحالات (إن وُجد إحداثي)
    const lastPos = await this.prisma.orderStatusHistory.findFirst({
      where: { orderId: order.id, latitude: { not: null } },
      orderBy: { createdAt: 'desc' },
      select: { latitude: true, longitude: true, createdAt: true },
    });

    return {
      data: {
        orderNumber: order.orderNumber,
        status: order.status,
        statusLabel: ORDER_STATUS_LABELS[order.status as keyof typeof ORDER_STATUS_LABELS] || order.status,
        service: order.service?.nameAr,
        scheduledDate: order.scheduledDate,
        scheduledSlot: order.scheduledSlot,
        from: { address: order.fromAddress, city: order.fromCity?.nameAr, latitude: Number(order.fromLatitude), longitude: Number(order.fromLongitude) },
        to: { address: order.toAddress, city: order.toCity?.nameAr, latitude: Number(order.toLatitude), longitude: Number(order.toLongitude) },
        driver: order.assignedTo ? { name: `${order.assignedTo.firstName} ${order.assignedTo.lastName}`, phone: order.assignedTo.phone, avatarUrl: order.assignedTo.avatarUrl } : null,
        driverLocation: lastPos ? { latitude: Number(lastPos.latitude), longitude: Number(lastPos.longitude), at: lastPos.createdAt } : null,
        createdAt: order.createdAt,
      },
    };
  }
}
```

```typescript
// apps/web/src/lib/api-client.ts
// عميل HTTP موحد: Bearer token + refresh تلقائي + طابور أثناء الـ refresh + أخطاء عربية.
// يسد F1 (تكرار), F7 (أخطاء), F8 (refresh), ويغذي F4.

export const API_URL = process.env.NEXT_PUBLIC_API_URL || 'http://localhost:4000/api/v1';

export interface ApiError {
  code: string;
  message: string;
  details?: Array<{ field?: string; message: string }>;
  requestId?: string;
}

export class ApiException extends Error {
  status: number;
  payload: ApiError;
  constructor(status: number, payload: ApiError) {
    super(payload?.message || `HTTP ${status}`);
    this.status = status;
    this.payload = payload;
  }
}

// ---- hooks مسجّلة من auth-store (تجنب circular import) ----
let getToken: () => string | null = () => null;
let setToken: (t: string | null) => void = () => {};
let onAuthFail: () => void = () => {};

export function configureApiClient(hooks: {
  getToken: () => string | null;
  setToken: (t: string | null) => void;
  onAuthFail: () => void;
}) {
  getToken = hooks.getToken;
  setToken = hooks.setToken;
  onAuthFail = hooks.onAuthFail;
}

// ---- طابور الـ refresh (يمنع refresh متزامن متعدد) ----
let refreshPromise: Promise<string | null> | null = null;

async function refreshAccessToken(): Promise<string | null> {
  if (refreshPromise) return refreshPromise;
  refreshPromise = (async () => {
    try {
      // الـ refresh token في httpOnly cookie (credentials: include)
      const res = await fetch(`${API_URL}/auth/refresh`, { method: 'POST', credentials: 'include' });
      if (!res.ok) return null;
      const json = await res.json();
      const newToken = json?.accessToken || json?.data?.accessToken || null;
      if (newToken) setToken(newToken);
      return newToken;
    } catch {
      return null;
    } finally {
      refreshPromise = null;
    }
  })();
  return refreshPromise;
}

function isAuthEndpoint(url: string) {
  return /\/auth\/(login|register|refresh|forgot-password|reset-password)/.test(url);
}

async function parseError(res: Response): Promise<ApiError> {
  try {
    const json = await res.json();
    return json?.error || { code: 'HTTP_ERROR', message: `خطأ ${res.status}` };
  } catch {
    return { code: 'HTTP_ERROR', message: `خطأ ${res.status}` };
  }
}

export interface RequestOptions extends Omit<RequestInit, 'body'> {
  body?: any;
  skipAuth?: boolean;
}

/**
 * الطلب الأساسي. عند 401 (وليس على endpoint مصادقة) يحاول refresh مرة ثم يعيد.
 */
export async function apiRequest<T = any>(path: string, options: RequestOptions = {}): Promise<T> {
  const { body, skipAuth, headers, ...rest } = options;
  const url = path.startsWith('http') ? path : `${API_URL}${path.startsWith('/') ? '' : '/'}${path}`;

  const buildHeaders = (token: string | null) => ({
    'Content-Type': 'application/json',
    'Accept-Language': 'ar',
    ...(token && !skipAuth ? { Authorization: `Bearer ${token}` } : {}),
    ...(headers as Record<string, string>),
  });

  let res = await fetch(url, {
    ...rest,
    credentials: 'include', // للـ refresh cookie
    headers: buildHeaders(getToken()),
    body: body !== undefined ? (typeof body === 'string' ? body : JSON.stringify(body)) : undefined,
  });

  // 401 → حاول refresh مرة واحدة (ما عدا endpoints المصادقة نفسها)
  if (res.status === 401 && !isAuthEndpoint(url) && !skipAuth) {
    const newToken = await refreshAccessToken();
    if (newToken) {
      res = await fetch(url, {
        ...rest,
        credentials: 'include',
        headers: buildHeaders(newToken),
        body: body !== undefined ? (typeof body === 'string' ? body : JSON.stringify(body)) : undefined,
      });
    } else {
      onAuthFail(); // لا يمكن الإنعاش → logout + توجيه
      throw new ApiException(401, { code: 'SESSION_EXPIRED', message: 'انتهت الجلسة. سجّل دخولك مجدداً.' });
    }
  }

  if (!res.ok) throw new ApiException(res.status, await parseError(res));

  if (res.status === 204) return undefined as T;
  const contentType = res.headers.get('content-type') || '';
  return contentType.includes('application/json') ? (res.json() as Promise<T>) : (res.text() as any);
}

// اختصارات مريحة
export const api = {
  get: <T = any>(path: string, opts?: RequestOptions) => apiRequest<T>(path, { ...opts, method: 'GET' }),
  post: <T = any>(path: string, body?: any, opts?: RequestOptions) => apiRequest<T>(path, { ...opts, method: 'POST', body }),
  patch: <T = any>(path: string, body?: any, opts?: RequestOptions) => apiRequest<T>(path, { ...opts, method: 'PATCH', body }),
  del: <T = any>(path: string, opts?: RequestOptions) => apiRequest<T>(path, { ...opts, method: 'DELETE' }),
};

/** يحوّل خطأً مجهولاً إلى رسالة عربية قابلة للعرض في toast */
export function toErrorMessage(err: unknown, fallback = 'حدث خطأ، حاول مجدداً'): string {
  if (err instanceof ApiException) {
    if (err.payload?.details?.length) {
      return err.payload.details.map((d) => d.message).join(' • ');
    }
    return err.payload?.message || err.message || fallback;
  }
  if (err instanceof Error) return err.message || fallback;
  return fallback;
}
```

### 📁 الملف 3: `apps/web/src/lib/auth-store.ts` (محدّث — يسد F4, F8)

```typescript
// apps/web/src/lib/auth-store.ts
// يحل محل النسخة المرجعية: apiClient + MFA flow + logout بـ refreshToken + تهيئة.
import { create } from 'zustand';
import { persist } from 'zustand/middleware';
import { api, configureApiClient, toErrorMessage } from './api-client';

export interface User {
  id: string; firstName: string; lastName: string;
  email: string; phone: string; role: string;
  avatarUrl?: string | null; mfaEnabled?: boolean;
  city?: { id: string; nameAr: string; slug: string } | null;
}

interface AuthState {
  user: User | null;
  accessToken: string | null;
  isAuthenticated: boolean;
  isLoading: boolean;
  // MFA
  mfaPending: boolean;
  tempToken: string | null;

  login: (identifier: string, password: string) => Promise<{ requiresMfa?: boolean }>;
  verifyMfa: (token: string) => Promise<void>;
  register: (data: RegisterInput) => Promise<void>;
  logout: () => Promise<void>;
  refreshUser: () => Promise<void>;
  initialize: () => Promise<void>;
  _setAccessToken: (t: string | null) => void;
}

interface RegisterInput {
  firstName: string; lastName: string; email: string; phone: string; password: string; cityId?: string;
}

export const useAuthStore = create<AuthState>()(
  persist(
    (set, get) => ({
      user: null, accessToken: null, isAuthenticated: false, isLoading: false,
      mfaPending: false, tempToken: null,

      login: async (identifier, password) => {
        set({ isLoading: true });
        try {
          const res = await api.post<any>('/auth/login', { identifier, password });
          if (res.requiresMfa) {
            set({ mfaPending: true, tempToken: res.tempToken, isLoading: false });
            return { requiresMfa: true };
          }
          set({ user: res.user, accessToken: res.accessToken, isAuthenticated: true, isLoading: false, mfaPending: false, tempToken: null });
          return {};
        } catch (e) { set({ isLoading: false }); throw e; }
      },

      verifyMfa: async (token) => {
        const tempToken = get().tempToken;
        if (!tempToken) throw new Error('لا يوجد رمز مؤقت. ابدأ تسجيل الدخول مجدداً.');
        set({ isLoading: true });
        try {
          const res = await api.post<any>('/auth/mfa/verify', { tempToken, token });
          set({ user: res.user, accessToken: res.accessToken, isAuthenticated: true, isLoading: false, mfaPending: false, tempToken: null });
        } catch (e) { set({ isLoading: false }); throw e; }
      },

      register: async (data) => {
        set({ isLoading: true });
        try {
          const res = await api.post<any>('/auth/register', data);
          set({ user: res.user, accessToken: res.accessToken, isAuthenticated: true, isLoading: false });
        } catch (e) { set({ isLoading: false }); throw e; }
      },

      logout: async () => {
        // F8: أمرر refreshToken (في cookie) + أنظّف الحالة
        try { await api.post('/auth/logout', {}); } catch {}
        set({ user: null, accessToken: null, isAuthenticated: false, mfaPending: false, tempToken: null });
      },

      refreshUser: async () => {
        try {
          const res = await api.get<any>('/auth/me');
          set({ user: res, isAuthenticated: true });
        } catch { set({ user: null, accessToken: null, isAuthenticated: false }); }
      },

      initialize: async () => {
        // عند تحميل الصفحة: إن وُجد token، تحقق من صلاحيته
        if (!get().accessToken) return;
        try {
          const res = await api.get<any>('/auth/me');
          set({ user: res, isAuthenticated: true });
        } catch {
          // apiClient حاول refresh بالفعل؛ إن فشل، onAuthFail نظّف الحالة
          set({ user: null, accessToken: null, isAuthenticated: false });
        }
      },

      _setAccessToken: (t) => set({ accessToken: t, isAuthenticated: !!t }),
    }),
    { name: 'motanaqil-auth', partialize: (s) => ({ accessToken: s.accessToken, user: s.user, isAuthenticated: s.isAuthenticated }) },
  ),
);

// ربط apiClient بالـ store (يُستدعى مرة واحدة من Provider)
configureApiClient({
  getToken: () => useAuthStore.getState().accessToken,
  setToken: (t) => useAuthStore.getState()._setAccessToken(t),
  onAuthFail: () => { useAuthStore.setState({ user: null, accessToken: null, isAuthenticated: false }); },
});

export { toErrorMessage };
```

```typescript
// apps/web/src/hooks/use-api.ts
// hook خفيف لجلب بيانات بدون مكتبة خارجية (يستخدم apiClient).
'use client';
import { useEffect, useState, useCallback, useRef } from 'react';
import { api, toErrorMessage } from '@/lib/api-client';

export function useApi<T = any>(path: string | null, deps: any[] = []) {
  const [data, setData] = useState<T | null>(null);
  const [loading, setLoading] = useState(!!path);
  const [error, setError] = useState<string | null>(null);
  const mounted = useRef(true);

  const reload = useCallback(async () => {
    if (!path) return;
    setLoading(true); setError(null);
    try {
      const res = await api.get<T>(path);
      if (mounted.current) setData(res);
    } catch (e) {
      if (mounted.current) setError(toErrorMessage(e, 'تعذّر تحميل البيانات'));
    } finally {
      if (mounted.current) setLoading(false);
    }
  // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [path, ...deps]);

  useEffect(() => { mounted.current = true; reload(); return () => { mounted.current = false; }; }, [reload]);
  return { data, loading, error, reload };
}
```

```typescript
// apps/web/src/lib/ws-client.ts
// عميل socket.io موحد: يمرر token، يعيد الاتصال بـ backoff، يعيد الاشتراك.
import { io, Socket } from 'socket.io-client';

const WS_BASE = process.env.NEXT_PUBLIC_WS_URL || process.env.NEXT_PUBLIC_API_URL?.replace('/api/v1', '') || 'http://localhost:4000';

let getToken: () => string | null = () => null;
export function setWsTokenGetter(fn: () => string | null) { getToken = fn; }

export function createSocket(namespace: '/notifications' | '/tracking'): Socket {
  const socket = io(`${WS_BASE}${namespace}`, {
    transports: ['websocket', 'polling'],
    auth: { token: getToken() },
    reconnection: true,
    reconnectionAttempts: Infinity,
    reconnectionDelay: 1000,
    reconnectionDelayMax: 15000,
  });

  // عند إعادة الاتصال، حدّث الـ token (قد يكون تغيّر بعد refresh)
  socket.io.on('reconnect_attempt', () => {
    (socket as any).auth = { token: getToken() };
  });

  socket.on('connect_error', (err: any) => {
    // إن كان الخطأ مصادقة، لا تُغرق الـ logs؛ الـ store سيتعامل عبر refresh عند الحاجة
    if (err?.message?.includes('Unauthorized') || err?.message?.includes('auth')) {
      socket.disconnect();
    }
  });

  return socket;
}
```

```typescript
'use client';
import { useEffect, useState, useCallback, useRef } from 'react';
import { api, toErrorMessage } from '@/lib/api-client';
import { createSocket, setWsTokenGetter } from '@/lib/ws-client';
import { useAuthStore } from '@/lib/auth-store';

export interface AppNotification { id: string; type: string; titleAr: string; bodyAr: string; isRead: boolean; createdAt: string; }

export function useNotifications() {
  const isAuthenticated = useAuthStore((s) => s.isAuthenticated);
  const [notifications, setNotifications] = useState<AppNotification[]>([]);
  const [unreadCount, setUnreadCount] = useState(0);
  const socketRef = useRef<any>(null);

  useEffect(() => { setWsTokenGetter(() => useAuthStore.getState().accessToken); }, []);

  // جلب أولي + socket
  useEffect(() => {
    if (!isAuthenticated) return;
    api.get<any>('/notifications?limit=10')
      .then((res) => { setNotifications(res?.data || []); setUnreadCount(res?.meta?.unreadCount ?? 0); })
      .catch(() => {});

    const socket = createSocket('/notifications');
    socketRef.current = socket;
    socket.on('notification:new', (n: AppNotification) => { setNotifications((p) => [n, ...p]); setUnreadCount((c) => c + 1); });
    socket.on('notifications:unread_count', (c: number) => setUnreadCount(c));
    return () => socket.disconnect();
  }, [isAuthenticated]);

  const markAsRead = useCallback(async (id: string) => {
    try { await api.patch(`/notifications/${id}/read`); setNotifications((p) => p.map((n) => n.id === id ? { ...n, isRead: true } : n)); setUnreadCount((c) => Math.max(0, c - 1)); } catch {}
  }, []);

  const markAllAsRead = useCallback(async () => {
    try { await api.post('/notifications/mark-all-read'); setNotifications((p) => p.map((n) => ({ ...n, isRead: true }))); setUnreadCount(0); } catch {}
  }, []);

  return { notifications, unreadCount, markAsRead, markAllAsRead };
}
```

## 📁 الملف 14: `scripts/verify-frontend-contract.sh` (يثبت اتفاق Front↔Back)

```bash
#!/usr/bin/env bash
# scripts/verify-frontend-contract.sh
# يضرب كل endpoint يستدعيه الـ Frontend ويتحقق من status + الحقول المتوقعة.
# يثبت أن الـ Frontend والـ Backend متفقان على الـ contract (بدون متصفح).
set -u
BASE="${BASE_URL:-http://localhost:4000/api/v1}"
PASS=0; FAIL=0
G='\033[0;32m'; R='\033[0;31m'; N='\033[0m'
ok(){ echo -e "${G}✅ $1${N}"; PASS=$((PASS+1)); }
no(){ echo -e "${R}❌ $1${N}"; FAIL=$((FAIL+1)); }
has(){ echo "$2" | grep -q "$3" && ok "$1" || no "$1 (missing: $3)"; }

echo "🔗 Verifying Frontend↔Backend contract..."
echo "------------------------------------------------------------"

# login (يستدعيه auth-store)
L=$(curl -s -X POST "$BASE/auth/login" -H "Content-Type: application/json" -d '{"identifier":"customer@motanaqil.com","password":"Motanaqil@2026"}')
has "POST /auth/login returns accessToken" "$L" '"accessToken"'
has "POST /auth/login returns user" "$L" '"user"'
T=$(echo "$L" | grep -o '"accessToken":"[^"]*"' | head -1 | cut -d'"' -f4)
H="Authorization: Bearer $T"

# /auth/me (initialize + refreshUser)
M=$(curl -s "$BASE/auth/me" -H "$H"); has "GET /auth/me returns mfaEnabled" "$M" 'mfaEnabled'

# /services (StepServiceSelection)
SV=$(curl -s "$BASE/services"); has "GET /services is array-ish" "$SV" 'nameAr'
SVC=$(echo "$SV" | grep -o '"id":"[^"]*"' | head -1 | cut -d'"' -f4)

# /orders POST (request/page) — يتطلب Bearer (يثبت سد F2)
C=$(curl -s -w "\n%{http_code}" -X POST "$BASE/orders" -H "$H" -H "Content-Type: application/json" \
  -d "{\"serviceId\":\"$SVC\",\"fromAddress\":{\"address\":\"حي النرجس\",\"coordinates\":{\"latitude\":24.8044,\"longitude\":46.6297}},\"toAddress\":{\"address\":\"حي الملقا\",\"coordinates\":{\"latitude\":24.7916,\"longitude\":46.6222}},\"scheduledDate\":\"2026-12-30T10:00:00Z\",\"scheduledSlot\":\"morning\"}")
CODE=$(echo "$C"|tail -1); BODY=$(echo "$C"|sed '$d')
[ "$CODE" = "201" ] && ok "POST /orders with Bearer → 201 (F2 fixed)" || no "POST /orders ($CODE)"
ON=$(echo "$BODY" | grep -o '"orderNumber":"[^"]*"' | head -1 | cut -d'"' -f4)

# /orders GET (dashboard useApi)
O=$(curl -s "$BASE/orders?page=1&limit=5" -H "$H"); has "GET /orders returns pagination" "$O" 'pagination'

# /tracking/order/:n (track page — F3)
TR=$(curl -s "$BASE/tracking/order/$ON"); has "GET /tracking/order/:n returns statusLabel" "$TR" 'statusLabel'
has "GET /tracking/order/:n returns from/to" "$TR" '"from"'

# /notifications (NotificationBell/use-notifications)
N1=$(curl -s "$BASE/notifications?limit=10" -H "$H"); has "GET /notifications returns meta.unreadCount" "$N1" 'unreadCount'
N2=$(curl -s "$BASE/notifications/unread-count" -H "$H"); has "GET /notifications/unread-count" "$N2" 'unreadCount'
N3=$(curl -s "$BASE/notifications/preferences" -H "$H"); has "GET /notifications/preferences" "$N3" 'emailEnabled'

# /orders/calculate-price (StepReview live price)
PR=$(curl -s -X POST "$BASE/orders/calculate-price" -H "Content-Type: application/json" -d "{\"serviceId\":\"$SVC\",\"fromLatitude\":24.8,\"fromLongitude\":46.6,\"toLatitude\":24.79,\"toLongitude\":46.62}")
has "POST /orders/calculate-price returns totalAmount" "$PR" 'totalAmount'

# 401 بدون token على /orders (يثبت أن الـ guard يعمل وأن الـ frontend يحتاج token)
U=$(curl -s -o /dev/null -w "%{http_code}" "$BASE/orders"); [ "$U" = "401" ] && ok "GET /orders no-token → 401 (guard active)" || no "no-token guard ($U)"

# Zod عربي (F7) — حقل ناقص يرجع details
Z=$(curl -s -X POST "$BASE/orders" -H "$H" -H "Content-Type: application/json" -d '{}')
has "validation returns Arabic details" "$Z" 'البيانات المدخلة غير صالحة\|مطلوب\|VALIDATION_ERROR'

echo "------------------------------------------------------------"
echo -e "Result: ${G}$PASS passed${N}, ${R}$FAIL failed${N}"
[ "$FAIL" -eq 0 ] && echo -e "${G}🎉 Frontend contract fully satisfied by Backend!${N}" || echo -e "${R}⚠️  Contract gaps remain.${N}"
exit $FAIL
```
