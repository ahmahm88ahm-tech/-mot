# 🚀 التنفيذ الفعلي: Notifications Module إنتاجي (الحلقة الأخيرة في الـ Backend)

> **الفجوات الحقيقية التي أسدّها الآن** (بعد مراجعة الكود السابق بدقة):
> 1. `notifications.module.ts` غير موجود → لا شيء مربوط فعلياً.
> 2. لا `NotificationsController` → الـ `NotificationBell` في الـ frontend يستدعي `GET /notifications` و`PATCH /preferences` وهما **غير موجودين**.
> 3. لا `NotificationsGateway` نهائي → الـ socket في `/notifications` لا يستقبل `notification:new`.
> 4. الـ in-app notification لا يُكتب في جدول `notifications` أصلاً.
> 5. أسماء الأحداث غير متطابقة (`auth.login_new_device` لم يُطلق من `auth.service`).
> 6. لا rate-limiting فعلي لكل مستخدم/قناة، ولا `failed` logging نهائي في الـ workers.
> 7. `app.module.ts` / `main.ts` لا يربطان كل الـ modules + `EventEmitterModule`.
>
> بعد هذا الرد، **إنشاء طلب سيُنتج إشعار in-app حقيقي في DB + يبثّه WebSocket فوراً**، وتُكتب سجلات التسليم لكل قناة.

```typescript
/** Atomic increment with TTL — للـ rate limiting */
async increment(key: string, ttlSeconds: number): Promise<number> {
  try {
    const n = await this.client.incr(key);
    if (n === 1) await this.client.expire(key, ttlSeconds);
    return n;
  } catch {
    return 0; // fail-open: لا نمنع الإشعار إن فشل Redis
  }
}

/** Raw get (string) — للـ counters */
async getRaw(key: string): Promise<string | null> {
  try { return await this.client.get(key); } catch { return null; }
}
```

```typescript
// apps/api/src/modules/notifications/notifications.gateway.ts
import {
  WebSocketGateway, WebSocketServer, OnGatewayConnection, OnGatewayDisconnect,
  ConnectedSocket, OnModuleInit, Logger,
} from '@nestjs/websockets';
import { Server, Socket } from 'socket.io';
import { JwtService } from '@nestjs/jwt';
import { EventEmitter2, OnEvent } from '@nestjs/event-emitter';
import { PrismaService } from '../../prisma/prisma.service';

/**
 * WebSocket gateway for in-app notifications.
 * - يُصادق الـ client بالـ JWT عند الاتصال.
 * - ينضم لغرفة user:{id}.
 * - يستمع للحدث الداخلي internal:notification_created ويبثّه فوراً.
 *
 * الربط: NotificationService يكتب صف in-app → يطلق internal event → هنا نبثّه.
 * (فك الـ coupling: الـ service لا يعرف شيئاً عن الـ socket)
 */
@WebSocketGateway({ namespace: '/notifications', cors: { origin: process.env.CORS_ORIGIN?.split(',') || '*', credentials: true } })
export class NotificationsGateway implements OnGatewayConnection, OnGatewayDisconnect, OnModuleInit {
  @WebSocketServer() server!: Server;
  private readonly logger = new Logger(NotificationsGateway.name);

  constructor(
    private readonly jwtService: JwtService,
    private readonly prisma: PrismaService,
    private readonly eventEmitter: EventEmitter2,
  ) {}

  onModuleInit() {
    // استماع للحدث الداخلي من NotificationService
    this.eventEmitter.on('internal:notification_created', (payload: { userId: string; notification: any }) => {
      this.server.to(`user:${payload.userId}`).emit('notification:new', payload.notification);
      // أعد حساب unread بدقة (يعمل عبر multiple instances)
      this.prisma.notification.count({ where: { userId: payload.userId, isRead: false } })
        .then((c) => this.server.to(`user:${payload.userId}`).emit('notifications:unread_count', c))
        .catch(() => {});
    });
  }

  async handleConnection(client: Socket) {
    try {
      const token = client.handshake.auth?.token as string | undefined;
      if (!token) { this.logger.warn(`socket ${client.id}: no token`); client.disconnect(true); return; }

      const payload: any = this.jwtService.verify(token);
      if (payload.type && payload.type !== 'access') { client.disconnect(true); return; }

      const user = await this.prisma.user.findUnique({ where: { id: payload.sub }, select: { id: true, status: true } });
      if (!user || user.status !== 'ACTIVE') { client.disconnect(true); return; }

      client.data.userId = user.id;
      client.join(`user:${user.id}`);

      // أرسل unread count الأولي
      const unread = await this.prisma.notification.count({ where: { userId: user.id, isRead: false } });
      client.emit('notifications:unread_count', unread);
      this.logger.debug(`socket ${client.id} joined user:${user.id}`);
    } catch (err: any) {
      this.logger.warn(`socket ${client.id} auth failed: ${err.message}`);
      client.disconnect(true);
    }
  }

  handleDisconnect(client: Socket) {
    this.logger.debug(`socket ${client.id} disconnected`);
  }
}
```

```typescript
// apps/api/src/modules/notifications/notifications.controller.ts
import { Controller, Get, Patch, Post, Body, Param, Query, UseGuards, HttpCode, HttpStatus } from '@nestjs/common';
import { ApiTags, ApiBearerAuth, ApiOperation } from '@nestjs/swagger';
import { PrismaService } from '../../prisma/prisma.service';
import { JwtAuthGuard } from '../../common/guards/jwt-auth.guard';
import { CurrentUser, type AuthUser } from '../../common/decorators/current-user.decorator';

@ApiTags('Notifications')
@Controller({ path: 'notifications', version: '1' })
@UseGuards(JwtAuthGuard)
@ApiBearerAuth()
export class NotificationsController {
  constructor(private readonly prisma: PrismaService) {}

  @Get()
  @ApiOperation({ summary: 'قائمة إشعاراتي' })
  async list(
    @CurrentUser() user: AuthUser,
    @Query('page') page?: string,
    @Query('limit') limit?: string,
    @Query('isRead') isRead?: string,
  ) {
    const p = Math.max(1, parseInt(page || '1', 10) || 1);
    const l = Math.min(parseInt(limit || '20', 10) || 20, 50);
    const where: any = { userId: user.id };
    if (isRead === 'true') where.isRead = true;
    if (isRead === 'false') where.isRead = false;

    const [data, total, unreadCount] = await Promise.all([
      this.prisma.notification.findMany({ where, orderBy: { createdAt: 'desc' }, skip: (p - 1) * l, take: l }),
      this.prisma.notification.count({ where }),
      this.prisma.notification.count({ where: { userId: user.id, isRead: false } }),
    ]);

    return { data, pagination: { page: p, limit: l, total, totalPages: Math.ceil(total / l) }, meta: { unreadCount } };
  }

  @Get('unread-count')
  @ApiOperation({ summary: 'عدد الإشعارات غير المقروءة' })
  async unreadCount(@CurrentUser() user: AuthUser) {
    const count = await this.prisma.notification.count({ where: { userId: user.id, isRead: false } });
    return { unreadCount: count };
  }

  @Patch(':id/read')
  @HttpCode(HttpStatus.OK)
  @ApiOperation({ summary: 'تعليم إشعار كمقروء' })
  async markRead(@CurrentUser() user: AuthUser, @Param('id') id: string) {
    await this.prisma.notification.updateMany({ where: { id, userId: user.id }, data: { isRead: true, readAt: new Date() } });
    return { success: true };
  }

  @Post('mark-all-read')
  @HttpCode(HttpStatus.OK)
  @ApiOperation({ summary: 'تعليم الكل كمقروء' })
  async markAllRead(@CurrentUser() user: AuthUser) {
    const r = await this.prisma.notification.updateMany({ where: { userId: user.id, isRead: false }, data: { isRead: true, readAt: new Date() } });
    return { success: true, markedCount: r.count };
  }

  // ===== Preferences =====
  @Get('preferences')
  @ApiOperation({ summary: 'تفضيلات الإشعارات' })
  async getPrefs(@CurrentUser() user: AuthUser) {
    let prefs = await this.prisma.notificationPreference.findUnique({ where: { userId: user.id } });
    if (!prefs) {
      prefs = await this.prisma.notificationPreference.create({ data: { userId: user.id } });
    }
    return { data: prefs };
  }

  @Patch('preferences')
  @ApiOperation({ summary: 'تحديث تفضيلات الإشعارات' })
  async updatePrefs(@CurrentUser() user: AuthUser, @Body() body: any) {
    const prefs = await this.prisma.notificationPreference.upsert({
      where: { userId: user.id },
      create: { userId: user.id, ...body },
      update: body,
    });
    return { data: prefs };
  }

  // ===== Push subscription =====
  @Post('push/subscribe')
  @HttpCode(HttpStatus.CREATED)
  @ApiOperation({ summary: 'تسجيل اشتراك Web Push' })
  async subscribePush(@CurrentUser() user: AuthUser, @Body() body: { endpoint: string; keys: { p256dh: string; auth: string }; userAgent?: string }) {
    const sub = await this.prisma.pushSubscription.upsert({
      where: { endpoint: body.endpoint },
      create: { userId: user.id, endpoint: body.endpoint, p256dh: body.keys.p256dh, auth: body.keys.auth, userAgent: body.userAgent, deviceType: 'web' },
      update: { userId: user.id, p256dh: body.keys.p256dh, auth: body.keys.auth, isActive: true },
    });
    return { data: { id: sub.id } };
  }

  @Post('push/unsubscribe')
  @HttpCode(HttpStatus.OK)
  async unsubscribePush(@CurrentUser() user: AuthUser, @Body() body: { endpoint: string }) {
    await this.prisma.pushSubscription.updateMany({ where: { endpoint: body.endpoint, userId: user.id }, data: { isActive: false } });
    return { success: true };
  }
}
```

## 📁 الملف 4: `apps/api/src/modules/notifications/notification.service.ts` (نهائي)

```typescript
// apps/api/src/modules/notifications/notification.service.ts
import { Injectable, Logger } from '@nestjs/common';
import { EventEmitter2 } from '@nestjs/event-emitter';
import { PrismaService } from '../../prisma/prisma.service';
import { CacheService } from '../../common/cache/cache.service';
import { NotificationQueueService } from './notification-queue.service';
import { TemplateEngineService } from './template-engine.service';
import type { DeliveryStatus } from '@prisma/client';

export interface NotificationRequest {
  userId: string;
  eventType: string;
  variables: Record<string, any>;
  channels?: string[];
  priority?: 'critical' | 'high' | 'normal' | 'low';
  eventId?: string;
}

// حدود الإرسال لكل مستخدم/قناة (حماية من الحلقات/الخلل)
const RATE_LIMITS: Record<string, { max: number; window: number }> = {
  email: { max: 30, window: 3600 },
  sms: { max: 15, window: 3600 },
  whatsapp: { max: 20, window: 3600 },
  push: { max: 60, window: 3600 },
};

@Injectable()
export class NotificationService {
  private readonly logger = new Logger(NotificationService.name);

  constructor(
    private readonly prisma: PrismaService,
    private readonly queueService: NotificationQueueService,
    private readonly templateEngine: TemplateEngineService,
    private readonly cache: CacheService,
    private readonly eventEmitter: EventEmitter2,
  ) {}

  async send(req: NotificationRequest): Promise<void> {
    const { userId, eventType, variables, priority = 'normal', eventId } = req;
    const corrId = eventId || `${eventType}-${userId}-${Date.now()}`;

    const prefs = await this.prisma.notificationPreference.findUnique({ where: { userId } });
    const isCritical = priority === 'critical';

    // quiet hours (تُطبق على القنوات الخارجية فقط؛ in-app دائماً)
    const inQuiet = !isCritical && this.isInQuietHours(prefs);

    const channels = req.channels || this.channelsForEvent(eventType, prefs);

    for (const channel of channels) {
      // 1) تفضيلات المستخدم
      if (!this.isChannelEnabled(channel, prefs, eventType)) {
        await this.log({ userId, eventType, channel, status: 'SKIPPED_PREFERENCE', eventId: corrId, recipient: '' });
        continue;
      }

      // 2) in-app: كتابة فورية + بث WebSocket (بدون queue/quiet/rate)
      if (channel === 'in_app') {
        await this.handleInApp(userId, eventType, variables, corrId);
        continue;
      }

      // 3) quiet hours
      if (inQuiet) {
        await this.log({ userId, eventType, channel, status: 'SKIPPED_QUIET_HOURS', eventId: corrId, recipient: '' });
        continue;
      }

      // 4) rate limit
      if (!(await this.passRateLimit(userId, channel))) {
        await this.log({ userId, eventType, channel, status: 'SKIPPED_RATE_LIMIT', eventId: corrId, recipient: '' });
        continue;
      }

      // 5) القالب + التصيير
      const template = await this.templateEngine.findTemplate(eventType, channel, 'ar');
      if (!template) { this.logger.warn(`no template ${eventType}/${channel}/ar`); continue; }
      const rendered = await this.templateEngine.render(template, variables);
      const recipient = await this.recipient(channel, userId);
      if (!recipient) { this.logger.warn(`no recipient for ${channel}/${userId}`); continue; }

      // 6) سجل QUEUED + طابور (الـ worker يحدّث نفس السجل عبر logId)
      const log = await this.log({ userId, eventType, channel, status: 'QUEUED', eventId: corrId, recipient, templateCode: template.code });
      await this.queueService.enqueue(channel, { logId: log.id, userId, eventType, eventId: corrId, templateCode: template.code, rendered, priority, recipient }, priority);
    }
  }

  /** إرسال مباشر يتجاوز التفضيلات (تنبيهات أمنية حرجة) */
  async sendDirect(userId: string, channel: string, content: { subject?: string; title: string; body: string; htmlBody?: string; ctaUrl?: string; ctaText?: string }): Promise<void> {
    if (channel === 'in_app') {
      const n = await this.prisma.notification.create({ data: { userId, type: 'direct', titleAr: content.title, bodyAr: content.body, channel: 'in_app' } });
      this.eventEmitter.emit('internal:notification_created', { userId, notification: n });
      return;
    }
    const recipient = await this.recipient(channel, userId);
    const log = await this.log({ userId, eventType: 'direct', channel, status: 'QUEUED', recipient });
    await this.queueService.enqueue(channel, { logId: log.id, userId, eventType: 'direct', rendered: content, priority: 'high', recipient }, 'high');
  }

  // ===== in-app =====
  private async handleInApp(userId: string, eventType: string, variables: Record<string, any>, corrId: string) {
    // للقالب in-app عنوان/نص؛ إن وُجد قالب نستخدمه، وإلا نص عام
    const template = await this.templateEngine.findTemplate(eventType, 'in_app', 'ar');
    const rendered = template ? await this.templateEngine.render(template, variables) : { title: eventType, body: '' };

    const n = await this.prisma.notification.create({
      data: { userId, type: eventType, titleAr: rendered.title, bodyAr: rendered.body, channel: 'in_app', data: variables },
    });
    await this.log({ userId, eventType, channel: 'in_app', status: 'SENT', eventId: corrId, recipient: userId, sentAt: new Date() });
    // بث فوري عبر الـ gateway
    this.eventEmitter.emit('internal:notification_created', { userId, notification: n });
  }

  // ===== helpers =====
  private channelsForEvent(eventType: string, prefs: any): string[] {
    const map: Record<string, string[]> = {
      'order.created': ['email', 'whatsapp', 'push', 'in_app'],
      'order.confirmed': ['sms', 'push', 'in_app'],
      'order.assigned': ['push', 'sms', 'in_app'],
      'order.completed': ['email', 'push', 'in_app'],
      'order.cancelled': ['sms', 'in_app'],
      'auth.login_new_device': ['email'],
      'auth.password_changed': ['email'],
    };
    return map[eventType] || ['in_app'];
  }

  private isChannelEnabled(channel: string, prefs: any, eventType: string): boolean {
    if (!prefs) return true;
    if (eventType.startsWith('auth.') || eventType.startsWith('security.')) return true; // أمنية دائماً
    if (eventType.startsWith('marketing.') && !prefs.promotions) return false;
    if (eventType.startsWith('order.') && !prefs.orderUpdates) return false;
    const key: Record<string, string> = { email: 'emailEnabled', sms: 'smsEnabled', whatsapp: 'whatsappEnabled', push: 'pushEnabled', in_app: 'inAppEnabled' };
    return (prefs as any)[key[channel]] ?? true;
  }

  private isInQuietHours(prefs: any): boolean {
    if (!prefs?.quietHoursEnabled) return false;
    const now = new Date();
    const t = now.toLocaleTimeString('en-GB', { hour12: false, hour: '2-digit', minute: '2-digit', timeZone: prefs.timezone || 'Asia/Riyadh' });
    const s = prefs.quietHoursStart || '22:00', e = prefs.quietHoursEnd || '08:00';
    return s < e ? (t >= s && t < e) : (t >= s || t < e); // يعبر منتصف الليل
  }

  private async passRateLimit(userId: string, channel: string): Promise<boolean> {
    const lim = RATE_LIMITS[channel];
    if (!lim) return true;
    const n = await this.cache.increment(`notif:rl:${userId}:${channel}`, lim.window);
    return n <= lim.max;
  }

  private async recipient(channel: string, userId: string): Promise<string> {
    if (channel === 'in_app') return userId;
    if (channel === 'push') {
      const s = await this.prisma.pushSubscription.findFirst({ where: { userId, isActive: true }, orderBy: { createdAt: 'desc' } });
      return s?.endpoint || '';
    }
    const u = await this.prisma.user.findUnique({ where: { id: userId }, select: { email: true, phone: true } });
    if (!u) return '';
    return channel === 'email' ? u.email : u.phone;
  }

  private log(d: { userId?: string; eventType?: string; channel: string; status: DeliveryStatus; eventId?: string; recipient?: string; templateCode?: string; sentAt?: Date }) {
    return this.prisma.notificationDeliveryLog.create({ data: d as any });
  }
}
```

## 📁 الملف 5: `apps/api/src/modules/notifications/notification-queue.service.ts` (نهائي)

```typescript
// apps/api/src/modules/notifications/notification-queue.service.ts
import { Injectable, Logger, OnModuleInit, OnModuleDestroy } from '@nestjs/common';
import { Queue, Worker, Job } from 'bullmq';
import Redis from 'ioredis';
import { PrismaService } from '../../prisma/prisma.service';
import { EmailProvider } from './providers/email.provider';
import { SmsProvider } from './providers/sms.provider';
import { WhatsAppProvider } from './providers/whatsapp.provider';
import { PushProvider } from './providers/push.provider';
import type { DeliveryStatus } from '@prisma/client';

const Q = { EMAIL: 'notif:email', SMS: 'notif:sms', WHATSAPP: 'notif:whatsapp', PUSH: 'notif:push' };
const PRIORITY: Record<string, number> = { critical: 1, high: 2, normal: 3, low: 4 };

@Injectable()
export class NotificationQueueService implements OnModuleInit, OnModuleDestroy {
  private readonly logger = new Logger(NotificationQueueService.name);
  private conn: Redis;
  private queues: Record<string, Queue> = {};
  private workers: Worker[] = [];

  constructor(
    private readonly prisma: PrismaService,
    private readonly email: EmailProvider,
    private readonly sms: SmsProvider,
    private readonly whatsapp: WhatsAppProvider,
    private readonly push: PushProvider,
  ) {
    this.conn = new Redis(process.env.REDIS_URL || 'redis://localhost:6379', { maxRetriesPerRequest: null });
  }

  onModuleInit() {
    Object.values(Q).forEach((name) => { this.queues[name] = new Queue(name, { connection: this.conn }); });
    this.workers.push(this.mkWorker(Q.EMAIL, 10, this.processEmail.bind(this)));
    this.workers.push(this.mkWorker(Q.SMS, 20, this.processSms.bind(this)));
    this.workers.push(this.mkWorker(Q.WHATSAPP, 15, this.processWhatsApp.bind(this)));
    this.workers.push(this.mkWorker(Q.PUSH, 50, this.processPush.bind(this)));
    this.logger.log('Notification queues + workers ready');
  }

  async onModuleDestroy() {
    await Promise.all(this.workers.map((w) => w.close()));
    await Promise.all(Object.values(this.queues).map((q) => q.close()));
    this.conn.disconnect();
  }

  async enqueue(channel: string, data: any, priority: string) {
    const name = Q[channel.toUpperCase() as keyof typeof Q];
    if (!name) return;
    await this.queues[name].add(channel, data, {
      priority: PRIORITY[priority] || 3,
      attempts: 5,
      backoff: { type: 'exponential', delay: 5000 },
      removeOnComplete: 200,
      removeOnFail: 1000,
    });
  }

  private mkWorker(name: string, concurrency: number, fn: (job: Job) => Promise<void>): Worker {
    const w = new Worker(name, async (job) => { await this.run(job, fn); }, { connection: this.conn, concurrency });
    w.on('failed', (job, err) => this.onFailed(job, err));
    return w;
  }

  /** غلاف مشترك: SENDING → تنفيذ → SENT، والفشل النهائي يُسجَّل في onFailed */
  private async run(job: Job, fn: (job: Job) => Promise<void>) {
    const logId = job.data.logId;
    if (logId) await this.updLog(logId, { status: 'SENDING', attempt: job.attemptsMade + 1 });
    await fn(job);
  }

  private async onFailed(job: Job | undefined, err: Error) {
    if (!job) return;
    const max = (job.opts.attempts as number) || 1;
    const isFinal = job.attemptsMade + 1 >= max;
    if (isFinal && job.data.logId) {
      await this.updLog(job.data.logId, { status: 'FAILED', errorMessage: err.message?.slice(0, 1000), failedAt: new Date(), attempt: job.attemptsMade + 1 });
      this.logger.error(`notif final fail [${job.name}] log=${job.data.logId}: ${err.message}`);
    }
  }

  private async updLog(id: string, data: Partial<{ status: DeliveryStatus; provider: string; providerMessageId: string; errorMessage: string; sentAt: Date; failedAt: Date; attempt: number }>) {
    try { await this.prisma.notificationDeliveryLog.update({ where: { id }, data: data as any }); } catch (e: any) { this.logger.warn(`log update fail ${id}: ${e.message}`); }
  }

  // ===== processors =====
  private async processEmail(job: Job) {
    const { rendered, recipient, logId } = job.data;
    const r = await this.email.send({ to: recipient, subject: rendered.subject || 'مُتنقِّل', html: rendered.htmlBody || `<p>${rendered.body}</p>`, text: rendered.body });
    if (r.status !== 'sent') throw new Error(r.error || 'email failed');
    await this.updLog(logId, { status: 'SENT', provider: r.provider, providerMessageId: r.messageId, sentAt: new Date() });
  }

  private async processSms(job: Job) {
    const { rendered, recipient, logId } = job.data;
    const r = await this.sms.send(recipient, rendered.body);
    if (r.status !== 'sent') throw new Error(r.error || 'sms failed');
    await this.updLog(logId, { status: 'SENT', provider: r.provider, providerMessageId: r.messageId, sentAt: new Date() });
  }

  private async processWhatsApp(job: Job) {
    const { rendered, recipient, logId, templateCode } = job.data;
    const r = templateCode
      ? await this.whatsapp.sendTemplate(recipient, templateCode, 'ar', [rendered.body])
      : await this.whatsapp.sendMessage(recipient, rendered.body);
    if (r.status !== 'sent') throw new Error(r.error || 'whatsapp failed');
    await this.updLog(logId, { status: 'SENT', provider: r.provider, providerMessageId: r.messageId, sentAt: new Date() });
  }

  private async processPush(job: Job) {
    const { rendered, recipient, logId, eventType } = job.data;
    const sub = await this.prisma.pushSubscription.findFirst({ where: { endpoint: recipient, isActive: true } });
    if (!sub) { await this.updLog(logId, { status: 'FAILED', errorMessage: 'subscription gone' }); return; }
    const r = await this.push.sendWebPush({ endpoint: sub.endpoint, p256dh: sub.p256dh, auth: sub.auth }, { title: rendered.title, body: rendered.body, url: rendered.ctaUrl, tag: eventType });
    if (r.status !== 'sent') throw new Error(r.error || 'push failed');
    await this.updLog(logId, { status: 'SENT', provider: r.provider, sentAt: new Date() });
  }
}
```
