## 1. جدول المستخدمين (users)

```sql
CREATE TABLE users (
    -- Primary Key
    id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    
    -- Personal Information
    national_id     VARCHAR(20) UNIQUE,
    first_name      VARCHAR(100) NOT NULL,
    last_name       VARCHAR(100) NOT NULL,
    full_name       VARCHAR(200) GENERATED ALWAYS AS (first_name || ' ' || last_name) STORED,
    date_of_birth   DATE,
    gender          VARCHAR(10) CHECK (gender IN ('male', 'female')),
    
    -- Contact Information
    email           VARCHAR(255) UNIQUE NOT NULL,
    email_verified  BOOLEAN DEFAULT FALSE,
    phone           VARCHAR(20) UNIQUE NOT NULL,
    phone_verified  BOOLEAN DEFAULT FALSE,
    whatsapp        VARCHAR(20),
    
    -- Credentials
    password_hash   VARCHAR(255) NOT NULL,
    two_factor_enabled BOOLEAN DEFAULT FALSE,
    last_password_change TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    
    -- Profile
    avatar_url      TEXT,
    bio             TEXT,
    
    -- Address
    address_line1   VARCHAR(255),
    address_line2   VARCHAR(255),
    district_id     UUID REFERENCES districts(id),
    city_id         UUID REFERENCES cities(id),
    region_id       UUID REFERENCES regions(id),
    postal_code     VARCHAR(10),
    latitude        DECIMAL(10, 8),
    longitude       DECIMAL(11, 8),
    
    -- Role & Status
    role            VARCHAR(30) NOT NULL DEFAULT 'customer'
                    CHECK (role IN ('customer', 'employee', 'driver', 'admin', 'manager', 'super_admin')),
    status          VARCHAR(20) DEFAULT 'active'
                    CHECK (status IN ('active', 'inactive', 'suspended', 'deleted')),
    
    -- Preferences
    preferred_language VARCHAR(5) DEFAULT 'ar',
    preferred_currency VARCHAR(3) DEFAULT 'SAR',
    notifications_enabled JSONB DEFAULT '{"email":true,"sms":true,"whatsapp":true,"push":true}',
    
    -- Metadata
    last_login_at   TIMESTAMP WITH TIME ZONE,
    last_login_ip   INET,
    login_count     INTEGER DEFAULT 0,
    
    -- Timestamps
    created_at      TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at      TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    deleted_at      TIMESTAMP WITH TIME ZONE
);

-- Indexes
CREATE INDEX idx_users_email ON users(email);
CREATE INDEX idx_users_phone ON users(phone);
CREATE INDEX idx_users_national_id ON users(national_id);
CREATE INDEX idx_users_city ON users(city_id);
CREATE INDEX idx_users_role ON users(role);
CREATE INDEX idx_users_status ON users(status);
CREATE INDEX idx_users_created_at ON users(created_at DESC);
CREATE INDEX idx_users_deleted_at ON users(deleted_at) WHERE deleted_at IS NOT NULL;

-- Comments
COMMENT ON TABLE users IS 'جدول المستخدمين - يخزن جميع بيانات العملاء والموظفين';
COMMENT ON COLUMN users.role IS 'دور المستخدم: customer, employee, driver, admin, manager, super_admin';
COMMENT ON COLUMN users.status IS 'حالة الحساب: active, inactive, suspended, deleted';
```

## 2. جدول الطلبات (orders)

```sql
CREATE TABLE orders (
    -- Primary Key
    id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    
    -- Order Identification
    order_number    VARCHAR(50) UNIQUE NOT NULL,  -- MTQ-2026-00001
    
    -- Relationships
    customer_id     UUID NOT NULL REFERENCES users(id),
    branch_id       UUID REFERENCES branches(id),
    service_id      UUID NOT NULL REFERENCES services(id),
    assigned_to     UUID REFERENCES users(id),
    
    -- From Location
    from_address    TEXT NOT NULL,
    from_district_id UUID REFERENCES districts(id),
    from_city_id    UUID REFERENCES cities(id),
    from_latitude   DECIMAL(10, 8) NOT NULL,
    from_longitude  DECIMAL(11, 8) NOT NULL,
    from_floor      INTEGER CHECK (from_floor >= 0),
    from_elevator   BOOLEAN DEFAULT FALSE,
    from_notes      TEXT,
    
    -- To Location
    to_address      TEXT NOT NULL,
    to_district_id  UUID REFERENCES districts(id),
    to_city_id      UUID REFERENCES cities(id),
    to_latitude     DECIMAL(10, 8) NOT NULL,
    to_longitude    DECIMAL(11, 8) NOT NULL,
    to_floor        INTEGER CHECK (to_floor >= 0),
    to_elevator     BOOLEAN DEFAULT FALSE,
    to_notes        TEXT,
    
    -- Route Information
    distance_km     DECIMAL(8, 2),
    duration_min    INTEGER,
    route_polyline  TEXT,
    
    -- Pricing
    base_price      DECIMAL(10, 2) NOT NULL,
    distance_price  DECIMAL(10, 2) DEFAULT 0,
    extra_services_price DECIMAL(10, 2) DEFAULT 0,
    discount_amount DECIMAL(10, 2) DEFAULT 0,
    tax_amount      DECIMAL(10, 2) NOT NULL,
    total_amount    DECIMAL(10, 2) NOT NULL,
    currency        VARCHAR(3) DEFAULT 'SAR',
    
    -- Status & Priority
    status          VARCHAR(30) NOT NULL DEFAULT 'created'
                    CHECK (status IN (
                        'created', 'confirmed', 'assigned',
                        'driver_started', 'arrived', 'loading',
                        'moving', 'unloading', 'installation',
                        'completed', 'cancelled', 'refunded'
                    )),
    priority        VARCHAR(20) DEFAULT 'normal'
                    CHECK (priority IN ('low', 'normal', 'high', 'urgent')),
    payment_status  VARCHAR(20) DEFAULT 'pending'
                    CHECK (payment_status IN ('pending', 'paid', 'partial', 'refunded')),
    payment_method  VARCHAR(30),
    
    -- Scheduling
    scheduled_date  TIMESTAMP WITH TIME ZONE,
    scheduled_slot  VARCHAR(20) CHECK (scheduled_slot IN ('morning', 'afternoon', 'evening')),
    
    -- Execution
    team_size       INTEGER DEFAULT 2 CHECK (team_size > 0),
    vehicle_type    VARCHAR(30) CHECK (vehicle_type IN ('small', 'medium', 'large', 'crane')),
    vehicle_plate   VARCHAR(20),
    
    -- Notes
    customer_notes  TEXT,
    internal_notes  TEXT,
    
    -- Review
    rating          INTEGER CHECK (rating BETWEEN 1 AND 5),
    review_text     TEXT,
    reviewed_at     TIMESTAMP WITH TIME ZONE,
    
    -- Cancellation
    cancelled_at    TIMESTAMP WITH TIME ZONE,
    cancellation_reason TEXT,
    cancelled_by    UUID REFERENCES users(id),
    
    -- Completion
    completed_at    TIMESTAMP WITH TIME ZONE,
    
    -- Metadata
    source          VARCHAR(30) DEFAULT 'web'
                    CHECK (source IN ('web', 'mobile', 'whatsapp', 'phone')),
    utm_source      VARCHAR(100),
    utm_medium      VARCHAR(100),
    utm_campaign    VARCHAR(100),
    ip_address      INET,
    user_agent      TEXT,
    
    -- Timestamps
    created_at      TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at      TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    deleted_at      TIMESTAMP WITH TIME ZONE
);

-- Indexes
CREATE INDEX idx_orders_number ON orders(order_number);
CREATE INDEX idx_orders_customer ON orders(customer_id);
CREATE INDEX idx_orders_assigned ON orders(assigned_to);
CREATE INDEX idx_orders_service ON orders(service_id);
CREATE INDEX idx_orders_branch ON orders(branch_id);
CREATE INDEX idx_orders_status ON orders(status);
CREATE INDEX idx_orders_priority ON orders(priority);
CREATE INDEX idx_orders_payment ON orders(payment_status);
CREATE INDEX idx_orders_scheduled ON orders(scheduled_date);
CREATE INDEX idx_orders_created ON orders(created_at DESC);
CREATE INDEX idx_orders_completed ON orders(completed_at DESC);
CREATE INDEX idx_orders_city ON orders(from_city_id, to_city_id);
CREATE INDEX idx_orders_source ON orders(source);
CREATE INDEX idx_orders_rating ON orders(rating) WHERE rating IS NOT NULL;

-- Comments
COMMENT ON TABLE orders IS 'جدول الطلبات - يخزن جميع طلبات نقل الأثاث';
COMMENT ON COLUMN orders.status IS 'حالة الطلب: created, confirmed, assigned, driver_started, arrived, loading, moving, unloading, installation, completed, cancelled, refunded';
COMMENT ON COLUMN orders.total_amount IS 'المبلغ الإجمالي شامل الضريبة';
```

```sql
    -- Status

    is_featured     BOOLEAN DEFAULT FALSE,

    gallery         JSONB DEFAULT '[]',
    
    -- Timestamps

    updated_at      TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    deleted_at      TIMESTAMP WITH TIME ZONE
```
