# 🚀 المرحلة (أ): توحيد Schema + Seed حقيقي + Storage + التحقق الفعلي

> **تفسير دقيق لطلبك:** "بيانات وهمية" تعني أحد أمرين: (1) responses مكتوبة يدوياً داخل الـ controller بدل قراءة DB، أو (2) seed عشوائي غير منطقي. الكود الذي كتبناه **يقرأ DB فعلاً** (لا hardcoded responses)، لذا التركيز هنا على ثلاثة أشياء تجعله "حقيقياً 100%":
> 1. **Schema موحّد واحد** يدمج كل الجداول بدون تكرار وبكل العلاقات الصحيحة.
> 2. **Seed منطقي حقيقي** (مدن/أحياء/خدمات/قوالب إشعارات/مستخدمون بكلمات مرور مشفّرة فعلياً) — بدونه لا يعمل أي endpoint.
> 3. **StorageService حقيقي** (S3 في الإنتاج + كتابة فعلية محلياً في التطوير) بدل الـ placeholder.
>
> ثم **سكربت تحقق** يضرب الـ API الحقيقية بسلسلة طلبات ويطبع ✅/❌ — ليثبت أن الربط يعمل فعلاً.

```prisma
model Order {
  id          String        @id @default(cuid())
  orderNumber String        @unique @map("order_number")

  customerId  String        @map("customer_id")
  customer    User          @relation("CustomerOrders", fields: [customerId], references: [id])
  assignedToId String?      @map("assigned_to_id")
  assignedTo  User?         @relation("DriverOrders", fields: [assignedToId], references: [id])
  serviceId   String        @map("service_id")
  service     Service       @relation(fields: [serviceId], references: [id])
  branchId    String?       @map("branch_id")
  branch      Branch?       @relation(fields: [branchId], references: [id])

  // From
  fromAddress     String   @map("from_address") @db.Text
  fromDistrictId  String?  @map("from_district_id")
  fromCityId      String?  @map("from_city_id")
  fromCity        City?    @relation("OrderFromCity", fields: [fromCityId], references: [id])
  fromLatitude    Decimal  @map("from_latitude") @db.Decimal(10, 8)
  fromLongitude   Decimal  @map("from_longitude") @db.Decimal(11, 8)
  fromFloor       Int?     @map("from_floor")
  fromHasElevator Boolean  @default(false) @map("from_has_elevator")
  fromNotes       String?  @map("from_notes") @db.Text

  // To
  toAddress     String   @map("to_address") @db.Text
  toDistrictId  String?  @map("to_district_id")
  toCityId      String?  @map("to_city_id")
  toCity        City?    @relation("OrderToCity", fields: [toCityId], references: [id])
  toLatitude    Decimal  @map("to_latitude") @db.Decimal(10, 8)
  toLongitude   Decimal  @map("to_longitude") @db.Decimal(11, 8)
  toFloor       Int?     @map("to_floor")
  toHasElevator Boolean  @default(false) @map("to_has_elevator")
  toNotes       String?  @map("to_notes") @db.Text

  // Route + Pricing
  distanceKm         Decimal?      @map("distance_km") @db.Decimal(8, 2)
  durationMin        Int?          @map("duration_min")
  basePrice          Decimal       @map("base_price") @db.Decimal(10, 2)
  distancePrice      Decimal       @default(0) @map("distance_price") @db.Decimal(10, 2)
  extraServicesPrice Decimal       @default(0) @map("extra_services_price") @db.Decimal(10, 2)
  discountAmount     Decimal       @default(0) @map("discount_amount") @db.Decimal(10, 2)
  taxAmount          Decimal       @map("tax_amount") @db.Decimal(10, 2)
  totalAmount        Decimal       @map("total_amount") @db.Decimal(10, 2)
  currency           String        @default("SAR")

  // Status + Schedule + Execution
  status        OrderStatus   @default(CREATED)
  priority      OrderPriority @default(NORMAL)
  paymentStatus PaymentStatus @default(PENDING) @map("payment_status")
  scheduledDate DateTime?     @map("scheduled_date")
  scheduledSlot String?       @map("scheduled_slot")
  teamSize      Int           @default(2) @map("team_size")
  vehicleType   String?       @map("vehicle_type")

  customerNotes  String?    @map("customer_notes") @db.Text
  internalNotes  String?    @map("internal_notes") @db.Text
  rating         Int?
  reviewText     String?    @map("review_text") @db.Text
  reviewedAt     DateTime?  @map("reviewed_at")
  cancelledAt    DateTime?  @map("cancelled_at")
  cancellationReason String? @map("cancellation_reason") @db.Text
  cancelledBy    String?    @map("cancelled_by")
  completedAt    DateTime?  @map("completed_at")

  source     String  @default("web")
  ipAddress  String? @map("ip_address")

  images        OrderImage[]
  statusHistory OrderStatusHistory[]
  reviews       Review[]

  createdAt DateTime  @default(now()) @map("created_at")
  updatedAt DateTime  @updatedAt @map("updated_at")
  deletedAt DateTime? @map("deleted_at")

  @@index([orderNumber])
  @@index([customerId])
  @@index([assignedToId])
  @@index([status])
  @@index([scheduledDate])
  @@index([createdAt])
  @@map("orders")
}
```

```typescript
// apps/api/src/common/storage/storage.service.ts
import { Injectable, Logger, OnModuleInit } from '@nestjs/common';
import { S3Client, PutObjectCommand, DeleteObjectCommand } from '@aws-sdk/client-s3';
import * as fs from 'fs/promises';
import * as path from 'path';

export interface StoredFile {
  url: string;
  thumbnailUrl?: string;
  key: string;
}

/**
 * Real storage service.
 * - Production (S3_* env set): uploads to S3/MinIO via AWS SDK.
 * - Development (no S3 creds): writes to ./public/uploads and serves via /uploads.
 * Both paths return a real, working URL — no placeholders.
 */
@Injectable()
export class StorageService implements OnModuleInit {
  private readonly logger = new Logger(StorageService.name);
  private s3?: S3Client;
  private bucket?: string;
  private useLocal = false;
  private localDir = path.resolve(process.cwd(), 'public', 'uploads');
  private publicBaseUrl: string;

  constructor() {
    this.publicBaseUrl = (process.env.PUBLIC_BASE_URL || 'http://localhost:4000').replace(/\/$/, '');
  }

  onModuleInit() {
    const endpoint = process.env.S3_ENDPOINT;
    const region = process.env.S3_REGION || 'me-central-1';
    const accessKey = process.env.S3_ACCESS_KEY;
    const secretKey = process.env.S3_SECRET_KEY;
    this.bucket = process.env.S3_BUCKET || 'motanaqil-uploads';

    if (endpoint && accessKey && secretKey) {
      this.s3 = new S3Client({
        endpoint, region, forcePathStyle: true,
        credentials: { accessKeyId: accessKey, secretAccessKey: secretKey },
      });
      this.logger.log(`Storage: S3 mode → bucket=${this.bucket}`);
    } else {
      this.useLocal = true;
      fs.mkdir(this.localDir, { recursive: true }).catch(() => {});
      this.logger.warn('Storage: LOCAL mode (no S3_* env). Files saved to ./public/uploads');
    }
  }

  /**
   * Store a buffer. `variants` lets caller pass pre-optimized webp/thumb buffers.
   */
  async store(variants: Array<{ buffer: Buffer; ext: string; suffix?: string; contentType: string }>): Promise<StoredFile> {
    const base = `${Date.now()}-${Math.random().toString(36).slice(2, 8)}`;
    let mainUrl = '';
    let mainKey = '';
    let thumbUrl: string | undefined;

    for (const v of variants) {
      const key = `uploads/${base}${v.suffix ? `-${v.suffix}` : ''}.${v.ext}`;
      const url = await this.put(key, v.buffer, v.contentType);
      if (!v.suffix) { mainUrl = url; mainKey = key; }
      else if (v.suffix === 'thumb') thumbUrl = url;
    }

    return { url: mainUrl, thumbnailUrl: thumbUrl, key: mainKey };
  }

  async delete(key: string): Promise<void> {
    try {
      if (this.useLocal) {
        await fs.unlink(path.join(process.cwd(), 'public', key)).catch(() => {});
      } else if (this.s3) {
        await this.s3.send(new DeleteObjectCommand({ Bucket: this.bucket, Key: key }));
      }
    } catch (err: any) {
      this.logger.warn(`Storage delete failed for ${key}: ${err.message}`);
    }
  }

  private async put(key: string, buffer: Buffer, contentType: string): Promise<string> {
    if (this.useLocal) {
      const fullPath = path.join(process.cwd(), 'public', key);
      await fs.mkdir(path.dirname(fullPath), { recursive: true });
      await fs.writeFile(fullPath, buffer);
      return `${this.publicBaseUrl}/${key}`;
    }
    await this.s3!.send(new PutObjectCommand({
      Bucket: this.bucket, Key: key, Body: buffer, ContentType: contentType,
      CacheControl: 'public, max-age=31536000, immutable',
    }));
    const endpoint = process.env.S3_ENDPOINT;
    return endpoint ? `${endpoint}/${this.bucket}/${key}` : `https://${this.bucket}.s3.${process.env.S3_REGION}.amazonaws.com/${key}`;
  }
}
```

```typescript
  const isValid = await this.imageProcessor.validateImage(file.buffer);
  if (!isValid) throw new Error('الملف ليس صورة صالحة');

  const variants = await this.imageProcessor.processUpload(file.buffer, file.originalname);

  // تخزين فعلي — يرجع URL يعمل فعلاً
  const stored = await this.storage.store([
    { buffer: variants.webp.buffer, ext: 'webp', contentType: 'image/webp' },
    { buffer: variants.thumbnail.buffer, ext: 'webp', suffix: 'thumb', contentType: 'image/webp' },
  ]);

  return this.prisma.media.create({
    data: {
      filename: stored.key.split('/').pop()!,
      originalName: file.originalname,
      mimeType: 'image/webp',
      sizeBytes: variants.webp.sizeBytes,
      url: stored.url,
      thumbnailUrl: stored.thumbnailUrl,
      width: variants.webp.width,
      height: variants.webp.height,
      folder: 'uploads',
      uploadedBy: uploadedBy ?? null,
    },
  });
}
```

## 📁 الملف أ.6: `scripts/verify-api.sh` (يثبت الربط الحقيقي)

```bash
#!/usr/bin/env bash
# scripts/verify-api.sh
# يضرب الـ API الحقيقية بسلسلة طلبات ويثبت أن كل شيء مربوط بـ DB فعلياً.
# المتطلب: server شغّال على localhost:4000 + DB مُهيّأة بـ seed.
set -u
BASE="${BASE_URL:-http://localhost:4000/api/v1}"
PASS=0; FAIL=0
GREEN='\033[0;32m'; RED='\033[0;31m'; NC='\033[0m'

check() { # $1=name $2=expected_status $3=actual_status
  if [ "$2" = "$3" ]; then echo -e "${GREEN}✅ $1${NC} ($3)"; PASS=$((PASS+1));
  else echo -e "${RED}❌ $1${NC} (expected $2, got $3)"; FAIL=$((FAIL+1)); fi
}

echo "🔍 Verifying MOTANAQIL API against real DB..."
echo "   Base: $BASE"
echo "------------------------------------------------------------"

# 1) Health
S=$(curl -s -o /dev/null -w "%{http_code}" "$BASE/../health"); check "Health endpoint" 200 "$S"

# 2) Services (يقرأ من DB المزروعة)
S=$(curl -s -o /dev/null -w "%{http_code}" "$BASE/services"); check "GET /services (from DB)" 200 "$S"

# 3) Cities
S=$(curl -s -o /dev/null -w "%{http_code}" "$BASE/cities"); check "GET /cities (from DB)" 200 "$S"

# 4) Login customer (يقرأ user المزروع + يتحقق bcrypt)
LOGIN=$(curl -s -w "\n%{http_code}" -X POST "$BASE/auth/login" \
  -H "Content-Type: application/json" \
  -d '{"identifier":"customer@motanaqil.com","password":"Motanaqil@2026"}')
CODE=$(echo "$LOGIN" | tail -1); BODY=$(echo "$LOGIN" | sed '$d')
check "POST /auth/login (bcrypt verify)" 200 "$CODE"
TOKEN=$(echo "$BODY" | grep -o '"accessToken":"[^"]*"' | head -1 | cut -d'"' -f4)
[ -n "$TOKEN" ] && echo -e "${GREEN}   ↳ access token received${NC}" || echo -e "${RED}   ↳ no token!${NC}"

# 5) /auth/me (يستخدم الـ JWT + JwtStrategy)
S=$(curl -s -o /dev/null -w "%{http_code}" "$BASE/auth/me" -H "Authorization: Bearer $TOKEN")
check "GET /auth/me (JWT guard)" 200 "$S"

# 6) Create order (يكتب DB + state machine + pricing)
SVC=$(curl -s "$BASE/services" | grep -o '"id":"[^"]*"' | head -1 | cut -d'"' -f4)
CREATE=$(curl -s -w "\n%{http_code}" -X POST "$BASE/orders" \
  -H "Authorization: Bearer $TOKEN" -H "Content-Type: application/json" \
  -d "{\"serviceId\":\"$SVC\",\"fromAddress\":{\"address\":\"حي النرجس\",\"coordinates\":{\"latitude\":24.8044,\"longitude\":46.6297}},\"toAddress\":{\"address\":\"حي الملقا\",\"coordinates\":{\"latitude\":24.7916,\"longitude\":46.6222}},\"scheduledDate\":\"2026-12-30T10:00:00Z\",\"scheduledSlot\":\"morning\"}")
CODE=$(echo "$CREATE" | tail -1); check "POST /orders (write DB)" 201 "$CODE"

# 7) List my orders (يقرأ DB)
S=$(curl -s -o /dev/null -w "%{http_code}" "$BASE/orders" -H "Authorization: Bearer $TOKEN")
check "GET /orders (read DB)" 200 "$S"

# 8) Calculate price (pure logic + DB service)
S=$(curl -s -o /dev/null -w "%{http_code}" -X POST "$BASE/orders/calculate-price" \
  -H "Content-Type: application/json" \
  -d "{\"serviceId\":\"$SVC\",\"fromLatitude\":24.80,\"fromLongitude\":46.62,\"toLatitude\":24.79,\"toLongitude\":46.62}")
check "POST /orders/calculate-price" 200 "$S"

# 9) CMS pages (يقرأ DB المزروعة)
S=$(curl -s -o /dev/null -w "%{http_code}" "$BASE/cms/pages/%D9%85%D9%86-%D9%86%D8%AD%D9%86")
check "GET /cms/pages/من-نحن (from DB)" 200 "$S"

# 10) CMS faqs
S=$(curl -s -o /dev/null -w "%{http_code}" "$BASE/cms/faqs"); check "GET /cms/faqs (from DB)" 200 "$S"

# 11) Public settings
S=$(curl -s -o /dev/null -w "%{http_code}" "$BASE/cms/settings/public"); check "GET /cms/settings/public" 200 "$S"

# 12) Validation rejection (Zod)
S=$(curl -s -o /dev/null -w "%{http_code}" -X POST "$BASE/orders" \
  -H "Authorization: Bearer $TOKEN" -H "Content-Type: application/json" -d '{}')
check "POST /orders {} → validation 422" 422 "$S"

# 13) Auth guard rejection
S=$(curl -s -o /dev/null -w "%{http_code}" "$BASE/orders"); check "GET /orders no-token → 401" 401 "$S"

echo "------------------------------------------------------------"
echo -e "Result: ${GREEN}$PASS passed${NC}, ${RED}$FAIL failed${NC}"
[ "$FAIL" -eq 0 ] && echo -e "${GREEN}🎉 All endpoints wired to real DB!${NC}" || echo -e "${RED}⚠️  Some checks failed.${NC}"
exit $FAIL
```
