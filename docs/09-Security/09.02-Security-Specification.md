## 1. الرؤية والمبادئ الأمنية

### 1.1 المبادئ الحاكمة

| المبدأ | التطبيق في MOTANAQIL |
|--------|---------------------|
| **Security by Design** | الأمان مُدمج في كل مرحلة تصميم، وليس إضافة لاحقة |
| **Defense in Depth** | طبقات حماية متعددة، فشل واحدة لا يُسقط النظام |
| **Zero Trust** | لا ثقة افتراضية، تحقق دائماً، امنح أقل صلاحية |
| **Least Privilege** | كل مستخدم/خدمة يحصل على الحد الأدنى من الصلاحيات |
| **Fail Securely** | عند الفشل، النظام يغلق لا يفتح (default deny) |
| **Separation of Duties** | لا شخص واحد يتحكم بعملية حرجة كاملة |
| **Encrypt Everything** | تشفير البيانات في الحركة والسكون |
| **Audit Everything** | كل إجراء حساس مُسجّل ولا يُمحى |
| **Assume Breach** | نفترض الاختراق ونستعد له مسبقاً |

### 4.2 تنفيذ MFA (Multi-Factor Authentication)

```typescript
// auth/mfa.service.ts
import { authenticator } from 'otplib';
import * as QRCode from 'qrcode';

@Injectable()
export class MFAService {
  /**
   * توليد سر MFA للمستخدم
   */
  async generateSecret(userId: string): Promise<MFASetup> {
    const secret = authenticator.generateSecret();
    const otpAuthUrl = authenticator.keyuri(
      userId,
      'MOTANAQIL',
      secret
    );
    
    // تخزين السر مشفراً (لا يُخزن نصاً صريحاً)
    const encryptedSecret = await this.encrypt(secret);
    await this.prisma.userMFA.upsert({
      where: { userId },
      create: { userId, secretEncrypted: encryptedSecret, enabled: false },
      update: { secretEncrypted: encryptedSecret },
    });
    
    const qrCode = await QRCode.toDataURL(otpAuthUrl);
    
    return { secret, qrCode, otpAuthUrl };
  }

  /**
   * تفعيل MFA بعد التحقق الأول
   */
  async enable(userId: string, token: string): Promise<void> {
    const mfa = await this.prisma.userMFA.findUnique({ where: { userId } });
    const secret = await this.decrypt(mfa.secretEncrypted);
    
    if (!authenticator.verify({ token, secret })) {
      throw new UnauthorizedException('رمز التحقق غير صحيح');
    }
    
    // توليد رموز احتياطية (Backup codes)
    const backupCodes = this.generateBackupCodes(10);
    const hashedCodes = await Promise.all(
      backupCodes.map(c => this.hash(c))
    );
    
    await this.prisma.userMFA.update({
      where: { userId },
      data: {
        enabled: true,
        backupCodes: hashedCodes,
        enabledAt: new Date(),
      },
    });
    
    // عرض الرموز الاحتياطية مرة واحدة فقط
    return { backupCodes };
  }

  /**
   * التحقق من رمز MFA عند تسجيل الدخول
   */
  async verify(userId: string, token: string): Promise<boolean> {
    const mfa = await this.prisma.userMFA.findUnique({ where: { userId } });
    if (!mfa?.enabled) return true; // MFA غير مفعّل
    
    const secret = await this.decrypt(mfa.secretEncrypted);
    
    // 1. محاولة TOTP
    if (authenticator.verify({ token, secret })) {
      return true;
    }
    
    // 2. محاولة رمز احتياطي
    const hashedToken = await this.hash(token);
    const backupCode = mfa.backupCodes.find(c => c === hashedToken);
    if (backupCode) {
      // حذف الرمز الاحتياطي بعد استخدامه (one-time)
      await this.prisma.userMFA.update({
        where: { userId },
        data: { backupCodes: mfa.backupCodes.filter(c => c !== hashedToken) },
      });
      return true;
    }
    
    return false;
  }

  private generateBackupCodes(count: number): string[] {
    return Array.from({ length: count }, () =>
      crypto.randomBytes(5).toString('hex').match(/.{1,5}/g).join('-')
    );
  }
}
```

### 4.4 حماية تسجيل الدخول

```typescript
// auth/login.guard.ts
@Injectable()
export class LoginProtectionService {
  /**
   * حماية ضد Brute Force + Credential Stuffing
   */
  async protectLogin(identifier: string, ip: string): Promise<void> {
    // 1. فحص قفل الحساب
    const failures = await this.redis.get(`login:fail:${identifier}`);
    if (failures && parseInt(failures) >= 5) {
      const lockUntil = await this.redis.get(`login:lock:${identifier}`);
      if (lockUntil && Date.now() < parseInt(lockUntil)) {
        throw new UnauthorizedException('الحساب مقفل مؤقتاً. حاول لاحقاً');
      }
    }
    
    // 2. فحص Rate limit على الـ IP
    const ipAttempts = await this.redis.incr(`login:ip:${ip}`);
    await this.redis.expire(`login:ip:${ip}`, 900); // 15 دقيقة
    if (ipAttempts > 20) {
      throw new TooManyRequestsException('محاولات كثيرة من هذا العنوان');
    }
    
    // 3. فحص Credential Stuffing (IP يستخدم حسابات كثيرة)
    const ipAccounts = await this.redis.scard(`login:ip:accounts:${ip}`);
    if (ipAccounts > 10) {
      await this.flagSuspicious(ip, 'credential_stuffing');
      throw new UnauthorizedException('نشاط مشبوه. تواصل مع الدعم');
    }
  }

  async recordFailure(identifier: string, ip: string): Promise<void> {
    const failures = await this.redis.incr(`login:fail:${identifier}`);
    await this.redis.expire(`login:fail:${identifier}`, 1800);
    await this.redis.sadd(`login:ip:accounts:${ip}`, identifier);
    
    if (failures >= 5) {
      await this.redis.set(
        `login:lock:${identifier}`,
        Date.now() + 30 * 60 * 1000,
        'EX', 1800
      );
      await this.notifySecurityTeam(identifier, ip);
    }
  }

  async recordSuccess(identifier: string, ip: string): Promise<void> {
    await this.redis.del(`login:fail:${identifier}`);
    // تسجيل الجهاز الجديد
    await this.trackDevice(identifier, ip);
  }
}
```

```typescript
@Injectable()
export class OwnershipGuard implements CanActivate {
  constructor(private prisma: PrismaService) {}

  async canActivate(context: ExecutionContext): Promise<boolean> {
    const request = context.switchToHttp().getRequest();
    const user = request.user;
    const resourceId = request.params.id;
    const resourceType = this.getResourceType(context);
    
    // المدراء يتجاوزون فحص الملكية
    if (['admin', 'super_admin', 'manager'].includes(user.role)) {
      return true;
    }
    
    // التحقق من الملكية
    const resource = await this.prisma[resourceType].findUnique({
      where: { id: resourceId },
    });
    
    if (!resource) {
      throw new NotFoundException('المورد غير موجود');
    }
    
    // هل المستخدم يملك هذا المورد؟
    const ownerId = resource.customerId || resource.userId;
    if (ownerId !== user.id) {
      // تسجيل محاولة IDOR (مهم أمنياً)
      await this.auditService.logIdorAttempt(user.id, resourceType, resourceId);
      throw new ForbiddenException('ليس لديك صلاحية الوصول لهذا المورد');
    }
    
    return true;
  }
}
```

### 6.2 Refresh Token Rotation + Reuse Detection

```typescript
// auth/refresh.service.ts
@Injectable()
export class RefreshTokenService {
  /**
   * تدوير الرمز مع كشف إعادة الاستخدام
   */
  async rotate(refreshToken: string): Promise<TokenPair> {
    const tokenHash = this.hash(refreshToken);
    const stored = await this.prisma.refreshToken.findUnique({
      where: { tokenHash },
    });
    
    if (!stored) {
      throw new UnauthorizedException('رمز غير صالح');
    }
    
    // 1. كشف إعادة الاستخدام (الرمز مستخدم سابقاً = سرقة)
    if (stored.revokedAt) {
      // اختراق محتمل! أبطل العائلة كاملة
      await this.revokeTokenFamily(stored.familyId);
      await this.alertSecurityTeam(stored.userId, 'token_reuse_detected');
      throw new UnauthorizedException('نشاط مشبوه. تم تسجيل خروجك من كل الأجهزة');
    }
    
    // 2. فحص الانتهاء
    if (stored.expiresAt < new Date()) {
      throw new UnauthorizedException('انتهت صلاحية الرمز');
    }
    
    // 3. إبطال الرمز الحالي
    await this.prisma.refreshToken.update({
      where: { id: stored.id },
      data: { revokedAt: new Date() },
    });
    
    // 4. إصدار زوج جديد (نفس العائلة)
    return this.issueTokenPair(stored.userId, stored.familyId);
  }

  /**
   * إبطال عائلة الرموز كاملة (عند كشف سرقة)
   */
  private async revokeTokenFamily(familyId: string): Promise<void> {
    await this.prisma.refreshToken.updateMany({
      where: { familyId, revokedAt: null },
      data: { revokedAt: new Date() },
    });
    // إبطال كل الجلسات النشطة للمستخدم
    await this.sessionService.revokeAllUserSessions(/* userId */);
  }
}
```

```typescript
@Injectable()
export class FieldEncryptionService {
  async encrypt(plaintext: string): Promise<string> {
    const key = await this.kms.getDataKey(); // من KMS
    const iv = crypto.randomBytes(12);
    const cipher = crypto.createCipheriv('aes-256-gcm', key, iv);
    const encrypted = Buffer.concat([
      cipher.update(plaintext, 'utf8'),
      cipher.final(),
    ]);
    const authTag = cipher.getAuthTag();
    // IV + AuthTag + Ciphertext
    return Buffer.concat([iv, authTag, encrypted]).toString('base64');
  }

  async decrypt(ciphertext: string): Promise<string> {
    const key = await this.kms.getDataKey();
    const buf = Buffer.from(ciphertext, 'base64');
    const iv = buf.subarray(0, 12);
    const authTag = buf.subarray(12, 28);
    const encrypted = buf.subarray(28);
    const decipher = crypto.createDecipheriv('aes-256-gcm', key, iv);
    decipher.setAuthTag(authTag);
    return decipher.update(encrypted) + decipher.final('utf8');
  }
}
```

### 11.1 Upload Security Pipeline

```typescript
// storage/upload-security.service.ts
@Injectable()
export class UploadSecurityService {
  private readonly ALLOWED_MIME = {
    image: ['image/jpeg', 'image/png', 'image/webp', 'image/gif'],
    document: ['application/pdf'],
  };
  private readonly MAX_SIZE = { image: 10 * 1024 * 1024, document: 5 * 1024 * 1024 };
  private readonly MAGIC_NUMBERS = {
    'image/jpeg': ['FFD8FF'],
    'image/png': ['89504E47'],
    'image/webp': ['52494646'],
    'image/gif': ['47494638'],
    'application/pdf': ['25504446'],
  };

  async validate(file: Express.Multer.File, category: 'image' | 'document') {
    // 1. فحص الامتداد (ضعيف وحده)
    const ext = extname(file.originalname).toLowerCase();
    
    // 2. فحص MIME المعلن
    if (!this.ALLOWED_MIME[category].includes(file.mimetype)) {
      throw new BadRequestException('نوع ملف غير مسموح');
    }
    
    // 3. فحص الحجم
    if (file.size > this.MAX_SIZE[category]) {
      throw new BadRequestException('حجم الملف كبير جداً');
    }
    
    // 4. فحص Magic Number (الأهم - يكشف التنكّر)
    const header = file.buffer.subarray(0, 8).toString('hex').toUpperCase();
    const validMagic = this.MAGIC_NUMBERS[file.mimetype]?.some(m => header.startsWith(m));
    if (!validMagic) {
      throw new BadRequestException('محتوى الملف لا يطابق نوعه');
    }
    
    // 5. فحص أبعاد الصور (منع zip bombs / malformed)
    if (category === 'image') {
      const meta = await sharp(file.buffer).metadata();
      if (!meta.width || !meta.height || meta.width > 8000 || meta.height > 8000) {
        throw new BadRequestException('أبعاد الصورة غير صالحة');
      }
    }
    
    // 6. فحص البرمجيات الخبيثة (ClamAV)
    const scanResult = await this.clamav.scan(file.buffer);
    if (scanResult.isInfected) {
      await this.alertSecurityTeam('malware_upload', { filename: file.originalname });
      throw new BadRequestException('الملف مرفوض');
    }
    
    // 7. إعادة تشفير الصورة (يزيل أي payload مخفي في metadata)
    if (category === 'image') {
      file.buffer = await this.reEncodeImage(file.buffer);
    }
    
    return true;
  }

  /**
   * إعادة ترميز الصورة تزيل EXIF والبيانات المخفية
   */
  private async reEncodeImage(buffer: Buffer): Promise<Buffer> {
    return sharp(buffer)
      .rotate() // تصحيح الاتجاه من EXIF ثم حذفه
      .withMetadata({}) // لا metadata
      .webp({ quality: 85 })
      .toBuffer();
  }
}
```

```typescript
// common/audit/audit.service.ts
@Injectable()
export class AuditService {
  /**
   * تسجيل إجراء حساس (غير قابل للتعديل)
   */
  async log(entry: AuditEntry): Promise<void> {
    const enriched = {
      ...entry,
      timestamp: new Date(),
      actorId: entry.actorId,
      actorRole: entry.actorRole,
      action: entry.action,
      resourceType: entry.resourceType,
      resourceId: entry.resourceId,
      oldValues: this.redact(entry.oldValues),  // تنقية PII
      newValues: this.redact(entry.newValues),
      ipAddress: entry.ipAddress,
      userAgent: entry.userAgent,
      requestId: entry.requestId,
      // توقيع السجل لمنع التعديل
      signature: await this.sign(entry),
    };
    
    // حفظ في DB + إرسال لـ SIEM
    await this.prisma.auditLog.create({ data: enriched });
    await this.siem.send(enriched);
  }

  // تنقية البيانات الحساسة من السجلات
  private redact(obj: any): any {
    if (!obj) return obj;
    const sensitive = ['password', 'national_id', 'card_number', 'cvv', 'otp', 'token'];
    const copy = { ...obj };
    for (const key of Object.keys(copy)) {
      if (sensitive.some(s => key.toLowerCase().includes(s))) {
        copy[key] = '[REDACTED]';
      }
    }
    return copy;
  }
}
```

```typescript
      .patch(`/api/v1/users/profile`)
```
