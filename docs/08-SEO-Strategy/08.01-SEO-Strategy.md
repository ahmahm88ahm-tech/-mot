أن تصبح MOTANAQIL **النتيجة الأولى** في Google Saudi Arabia لكل عمليات البحث المتعلقة بنقل الأثاث في المملكة خلال 12 شهراً من الإطلاق.

## 4. Programmatic SEO

### 4.1 الفلسفة

> **توليد آلاف الصفحات عالية الجودة تلقائياً** من قاعدة البيانات، كل صفحة فريدة ومفيدة ومخصصة لمدينة/حي/خدمة محددة.

### 4.2 Page Generator Architecture

```typescript
// lib/seo/page-generator.ts
@Injectable()
export class SEOPageGenerator {
  constructor(
    private readonly prisma: PrismaService,
    private readonly contentTemplates: ContentTemplatesService,
    private readonly aiContentService: AIContentService, // اختياري، مع مراجعة بشرية
  ) {}

  /**
   * توليد جميع الصفحات البرمجية
   */
  async generateAllPages(): Promise<GeneratedPage[]> {
    const pages: GeneratedPage[] = [];

    // 1. صفحات المدن
    const cities = await this.prisma.city.findMany({
      where: { isActive: true },
      include: { districts: true, region: true },
    });

    for (const city of cities) {
      pages.push(await this.generateCityPage(city));
    }

    // 2. صفحات الأحياء
    for (const city of cities) {
      for (const district of city.districts) {
        pages.push(await this.generateDistrictPage(city, district));
      }
    }

    // 3. صفحات Service × City
    const services = await this.prisma.service.findMany({
      where: { isActive: true },
    });

    for (const service of services) {
      for (const city of cities) {
        pages.push(await this.generateServiceCityPage(service, city));
      }
    }

    // 4. صفحات Service × City × District (للأحياء الكبيرة)
    for (const service of services) {
      for (const city of cities) {
        const majorDistricts = city.districts.filter(d => d.isMajor);
        for (const district of majorDistricts) {
          pages.push(
            await this.generateServiceCityDistrictPage(service, city, district)
          );
        }
      }
    }

    return pages;
  }

  /**
   * توليد صفحة مدينة
   */
  private async generateCityPage(city: City): Promise<GeneratedPage> {
    const stats = await this.getCityStats(city.id);
    const topServices = await this.getTopServicesInCity(city.id);
    const topDistricts = city.districts.slice(0, 20);
    const testimonials = await this.getCityTestimonials(city.id);
    const faqs = this.getCityFAQs(city);

    return {
      slug: `cities/${city.slug}`,
      template: 'city',
      data: {
        city,
        stats,
        topServices,
        topDistricts,
        testimonials,
        faqs,
      },
      seo: {
        title: `نقل عفش ${city.nameAr} | أسعار تبدأ من 300 ريال | مُتنقِّل`,
        description: `خدمة نقل عفش احترافية في ${city.nameAr} مع فك وتركيب الأثاث. ${stats.districtsCount} حي مخدوم، ${stats.ordersCount}+ طلب منجز، تقييم ${stats.avgRating}⭐. اطلب خدمتك الآن.`,
        keywords: [
          `نقل عفش ${city.nameAr}`,
          `شركة نقل اثاث ${city.nameAr}`,
          `فك وتركيب ${city.nameAr}`,
        ],
        canonical: `https://motanaqil.com/cities/${city.slug}`,
      },
      schema: this.buildCitySchema(city, stats),
    };
  }

  /**
   * توليد محتوى فريد لكل صفحة (لتجنب Duplicate Content)
   */
  private async generateUniqueContent(
    template: string,
    variables: Record<string, any>
  ): Promise<string> {
    // استخدام قوالب متعددة + متغيرات ديناميكية
    const templates = this.contentTemplates.getVariations(template);
    const selectedTemplate = templates[
      Math.floor(Math.random() * templates.length)
    ];

    // دمج البيانات الحقيقية (إحصائيات، تقييمات، أسعار)
    const content = this.renderTemplate(selectedTemplate, {
      ...variables,
      currentYear: new Date().getFullYear(),
      uniqueStats: await this.getUniqueStats(variables),
    });

    return content;
  }
}
```

```typescript
  const schema: WithContext<BreadcrumbList> = {
    '@context': 'https://schema.org',
    '@type': 'BreadcrumbList',
    itemListElement: items.map((item, index) => ({

    })),
  };

    <>
      <script
        type="application/ld+json"
        dangerouslySetInnerHTML={{ __html: JSON.stringify(schema) }}
      />
      <nav aria-label="Breadcrumb" className="breadcrumbs">
        <ol>
          {items.map((item, i) => (
            <li key={i}>
              {item.href ? (
                <a href={item.href}>{item.label}</a>
              ) : (
                <span aria-current="page">{item.label}</span>
              )}
              {i < items.length - 1 && <span> / </span>}
```

```typescript
// lib/seo/internal-linking.ts
export function generateInternalLinks(page: Page): InternalLink[] {
  const links: InternalLink[] = [];

  // 1. روابط للصفحات الأب (Breadcrumbs)
  if (page.parent) {
    links.push({ href: page.parent.url, text: page.parent.title, type: 'breadcrumb' });
  }

  // 2. روابط للصفحات الشقيقة
  const siblings = await getSiblingPages(page);
  siblings.forEach(s => {
    links.push({ href: s.url, text: s.title, type: 'sibling' });
  });

  // 3. روابط للصفحات ذات الصلة (Contextual)
  const related = await getRelatedPages(page);
  related.forEach(r => {
    links.push({ href: r.url, text: r.anchorText, type: 'contextual' });
  });

  // 4. روابط للمدن/الأحياء المجاورة
  if (page.type === 'district') {
    const nearby = await getNearbyDistricts(page);
    nearby.forEach(n => {
      links.push({ href: n.url, text: `نقل عفش حي ${n.name}`, type: 'geo' });
    });
  }

  // 5. روابط للخدمات ذات الصلة
  if (page.type === 'service') {
    const relatedServices = await getRelatedServices(page);
    relatedServices.forEach(s => {
      links.push({ href: s.url, text: s.name, type: 'service' });
    });
  }

  return links;
}

// قواعد Internal Linking:
// - 50-100 رابط داخلي في كل صفحة
// - Anchor text وصفي (وليس "اضغط هنا")
// - توزيع PageRank للصفحات المهمة
// - روابط من المحتوى (Contextual) أقوى من Footer
```

```typescript
import { NextResponse } from 'next/server';

export function middleware(request: Next.Request) {
  const { pathname } = request.nextUrl;
  const acceptLanguage = request.headers.get('accept-language') || 'ar';
  
  // إذا لم يكن هناك لغة في URL
  if (!pathname.startsWith('/en') && !pathname.startsWith('/ar')) {
    // الكشف التلقائي (اختياري)
    const detectedLang = acceptLanguage.startsWith('en') ? 'en' : 'ar';
    
    // للعربية، لا نضيف prefix (default)
    if (detectedLang === 'ar') {
      return NextResponse.next();
    }
    
    // للإنجليزية، نعيد التوجيه
    return NextResponse.redirect(new URL(`/en${pathname}`, request.url));
  }
}
```
