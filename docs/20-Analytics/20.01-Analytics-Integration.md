## 1. الرؤية وأهداف القياس

### 1.1 الغرض

نظام التحليلات في MOTANAQIL مصمم للإجابة على الأسئلة التجارية الحرجة:

- 🎯 **من أين يأتي العميل؟** (SEO، إعلانات، سوشيال، مباشر)
- 🎯 **ماذا يفعل في الموقع؟** (مسار، نقاط الخروج، التفاعل)
- 🎯 **لماذا لا يُكمل الطلب؟** (تحليل التخلي عن النموذج)
- 🎯 **ما هي الحملة الأكثر ربحية؟** (ROAS دقيق)
- 🎯 **أي صفحة تجلب طلبات فعلية؟** (وليس مجرد زيارات)
- 🎯 **ما هو LTV (القيمة العمرية) للعميل؟**

### 1.2 الأهداف الكمية

| المقياس | الهدف |
|--------|------|
| **تغطية الأحداث** | 100% من التفاعلات المهمة |
| **دقة التتبع** | > 98% (مع Server-Side) |
| **تجاوز Ad Blockers** | > 90% من الأحداث عبر S2S |
| **زمن تحميل السكربتات** | < 100ms (لا تؤثر على CWV) |
| **Privacy Compliance** | 100% (PDPL + GDPR) |

### 3.3 Analytics Service (الواجهة الموحدة)

```typescript
// lib/analytics/analytics.service.ts
'use client';

class AnalyticsService {
  private dataLayer: any[];
  private consent: ConsentState;
  private sessionId: string;
  private sessionNumber: number;

  constructor() {
    this.dataLayer = (window as any).dataLayer = (window as any).dataLayer || [];
    this.consent = this.getConsent();
    this.sessionId = this.getOrCreateSessionId();
    this.sessionNumber = this.getSessionNumber();
  }

  /**
   * دفع حدث للـ Data Layer
   * هذه الدالة الوحيدة التي يجب أن يستدعيها الكود
   */
  track(event: AnalyticsEvent): void {
    // 1. التحقق من الموافقة
    if (!this.canTrack(event)) {
      this.queueForLater(event); // نحفظه حتى يوافق المستخدم
      return;
    }

    // 2. إثراء الحدث بالسياق التلقائي
    const enrichedEvent = this.enrich(event);

    // 3. الدفع للـ dataLayer
    this.dataLayer.push(enrichedEvent);

    // 4. تسجيل في First-Party (للتحليل الداخلي + S2S)
    this.sendToFirstParty(enrichedEvent);

    // 5. Log في التطوير فقط
    if (process.env.NODE_ENV === 'development') {
      console.log('[Analytics]', enrichedEvent.event, enrichedEvent);
    }
  }

  /**
   * إثراء الحدث بالسياق التلقائي
   */
  private enrich(event: AnalyticsEvent): AnalyticsEvent {
    const pageInfo = this.getPageInfo();
    const deviceInfo = this.getDeviceInfo();
    const campaignInfo = this.getCampaignInfo();

    return {
      ...event,
      // سياق الصفحة
      page_location: pageInfo.location,
      page_path: pageInfo.path,
      page_title: pageInfo.title,
      page_language: pageInfo.language,
      // سياق المستخدم
      user_id: this.getUserId(),
      user_type: this.getUserType(),
      login_state: this.getLoginState(),
      // سياق الحملة
      ...campaignInfo,
      // سياق الجهاز
      ...deviceInfo,
      // سياق الوقت
      timestamp: new Date().toISOString(),
      timezone: 'Asia/Riyadh',
      // الجلسة
      session_id: this.sessionId,
      session_number: this.sessionNumber,
      // الموافقة
      consent_status: this.consent,
    };
  }

  /**
   * هل يمكن تتبع هذا الحدث حسب الموافقة؟
   */
  private canTrack(event: AnalyticsEvent): boolean {
    // الأحداث الوظيفية دائماً مسموحة
    if (this.isFunctionalEvent(event)) return true;
    
    // أحداث التحليلات تحتاج موافقة analytics
    if (this.isAnalyticsEvent(event)) return this.consent.analytics;
    
    // أحداث الإعلانات تحتاج موافقة advertising
    if (this.isAdvertisingEvent(event)) return this.consent.advertising;
    
    return true;
  }

  /**
   * إرسال للـ First-Party endpoint (للـ S2S tracking)
   */
  private async sendToFirstParty(event: AnalyticsEvent): Promise<void> {
    // نستخدم sendBeacon لضمان الإرسال حتى عند إغلاق الصفحة
    const payload = JSON.stringify(event);
    
    if (navigator.sendBeacon) {
      navigator.sendBeacon('/api/v1/track/event', payload);
    } else {
      // Fallback
      fetch('/api/v1/track/event', {
        method: 'POST',
        body: payload,
        keepalive: true,
      }).catch(() => {});
    }
  }

  // ===== دوال مساعدة مختصرة =====
  
  trackPageView(data: Partial<PageViewEvent>) {
    this.track({ event: 'page_view', event_category: 'page', ...data } as PageViewEvent);
  }

  trackClick(data: Partial<ClickEvent>) {
    this.track({ event: 'element_click', event_category: 'engagement', ...data } as ClickEvent);
  }

  trackOrder(data: Partial<OrderEvent>) {
    this.track({ event: 'order_event', event_category: 'order', ...data } as OrderEvent);
  }

  trackPurchase(data: Partial<PurchaseEvent>) {
    this.track({ event: 'purchase', event_category: 'conversion', ...data } as PurchaseEvent);
  }

  trackForm(data: Partial<FormEvent>) {
    this.track({ event: 'form_event', event_category: 'form', ...data } as FormEvent);
  }
}

// Singleton
export const analytics = new AnalyticsService();
```

### 3.4 استخدام Analytics في المكونات

```typescript
// مثال: تتبع زر الواتساب العائم
export function WhatsAppButton({ phoneNumber }: { phoneNumber: string }) {
  const handleClick = () => {
    analytics.trackClick({
      element_type: 'whatsapp',
      element_text: 'تواصل عبر واتساب',
      element_position: 'floating_button',
      event_label: phoneNumber,
    });
    
    // فتح واتساب
    window.open(`https://wa.me/${phoneNumber}`, '_blank');
  };

  return (
    <button onClick={handleClick} aria-label="تواصل عبر واتساب">
      <WhatsAppIcon />
    </button>
  );
}

// مثال: تتبع خطوات نموذج الطلب
export function RequestForm() {
  const [step, setStep] = useState(1);

  const handleNext = () => {
    analytics.trackOrder({
      order_action: 'step_completed',
      order_step: step,
      order_step_name: STEP_NAMES[step],
    });
    setStep(step + 1);
  };

  const handleSubmit = async (data: OrderData) => {
    analytics.trackForm({
      form_action: 'submit',
      form_name: 'request_service',
      form_id: 'request-form',
    });

    try {
      const order = await createOrder(data);
      
      analytics.trackOrder({
        order_action: 'submitted',
        order_id: order.id,
        order_number: order.orderNumber,
        service_name: order.service.nameAr,
        order_value: order.totalAmount,
        from_city: order.fromAddress.city,
        to_city: order.toAddress.city,
      });

      analytics.trackPurchase({
        transaction_id: order.orderNumber,
        value: order.totalAmount,
        currency: 'SAR',
        items: [{
          item_id: order.service.id,
          item_name: order.service.nameAr,
          item_category: order.service.category.nameAr,
          price: order.totalAmount,
          quantity: 1,
        }],
      });
    } catch (error) {
      analytics.trackForm({
        form_action: 'error',
        form_name: 'request_service',
        form_id: 'request-form',
        error_message: error.message,
      });
    }
  };

  // تتبع التخلي عن النموذج
  useEffect(() => {
    const handleBeforeUnload = () => {
      if (step > 1 && step < 6) {
        analytics.trackForm({
          form_action: 'abandon',
          form_name: 'request_service',
          form_step: step,
        });
      }
    };
    window.addEventListener('beforeunload', handleBeforeUnload);
    return () => window.removeEventListener('beforeunload', handleBeforeUnload);
  }, [step]);

  return <form>...</form>;
}
```

```typescript
@Injectable()
export class MetaCAPIservice {
  private readonly pixelId = process.env.META_PIXEL_ID;
  private readonly accessToken = process.env.META_CAPI_ACCESS_TOKEN;
  private readonly apiUrl = `https://graph.facebook.com/v18.0/${this.pixelId}/events`;

  /**
   * إرسال حدث عبر CAPI
   */
  async sendEvent(event: MetaCAPIEvent): Promise<void> {
    const payload = {
      data: [{
        event_name: event.eventName,
        event_time: Math.floor(Date.now() / 1000),
        event_id: event.eventId,  // للـ deduplication مع Pixel
        action_source: 'website',

        user_data: this.hashUserData({

          ln: event.lastName,
          ct: event.city,
          st: event.region,

          external_id: event.userId,
          client_ip_address: event.ipAddress,
          client_user_agent: event.userAgent,
          fbp: event.fbp,
          fbc: event.fbc,
        }),
        
        // بيانات الحدث
        custom_data: {
          currency: 'SAR',
          value: event.value,
          content_ids: event.contentIds,
          content_name: event.contentName,
          content_category: event.contentCategory,
          content_type: 'product',
          num_items: event.numItems,
          order_id: event.orderId,
        },
        
        // بيانات الصفحة
        event_source_url: event.pageUrl,
        opt_out: false,
      }],
      test_event_code: process.env.NODE_ENV === 'development' ? 'TEST12345' : undefined,
    };

    const response = await fetch(`${this.apiUrl}?access_token=${this.accessToken}`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(payload),
    });

    if (!response.ok) {
      this.logger.error('Meta CAPI failed', await response.text());
    }
  }

  /**
   * تجزئة بيانات المستخدم (SHA256) - إلزامي من Meta
   */
  private hashUserData(data: Record<string, any>): Record<string, any> {
    const hashed: Record<string, any> = {};
    for (const [key, value] of Object.entries(data)) {
      if (value && key !== 'client_ip_address' && key !== 'client_user_agent') {
        hashed[key] = this.sha256(String(value).toLowerCase().trim());
      } else {
        hashed[key] = value; // IP و UA بدون تجزئة
      }
    }
    return hashed;
  }

  private sha256(value: string): string {
    return createHash('sha256').update(value).digest('hex');
  }
}
```

```typescript
    };
```

```typescript
    await prisma.analyticsEvent.create({
      data: {
        eventName: enrichedEvent.event,
        sessionId: enrichedEvent.session_id,
        userId: enrichedEvent.user_id,
        eventData: enrichedEvent,
        ipAddress: enrichedEvent.server_ip,
        userAgent: enrichedEvent.server_user_agent,
      },
    });
    
    // 3. إرسال للمنصات عبر Server-Side (في الخلفية)
    await trackingQueue.add('dispatch-event', enrichedEvent, {
      priority: getEventPriority(enrichedEvent.event),
    });
    
    return NextResponse.json({ success: true });
  } catch (error) {
    return NextResponse.json({ success: false }, { status: 400 });
  }
}
```

### 11.2 Event Dispatcher (Queue Processor)

```typescript
// processors/event-dispatcher.processor.ts
@Processor('tracking-events')
export class EventDispatcherProcessor {
  constructor(
    private readonly ga4: GA4ServerService,
    private readonly metaCAPI: MetaCAPIservice,
    private readonly tiktokEAPI: TikTokEAPIservice,
    private readonly snapCAPI: SnapCAPIservice,
  ) {}

  @Process('dispatch-event')
  async dispatch(job: Job<AnalyticsEvent>) {
    const event = job.data;
    
    // إرسال متوازي للمنصات المعنية
    const promises: Promise<any>[] = [];
    
    // GA4 Measurement Protocol (Server)
    if (this.shouldSendToGA4(event)) {
      promises.push(this.ga4.sendEvent(event).catch(e => this.logError('GA4', e)));
    }
    
    // Meta CAPI (للأحداث التجارية)
    if (this.shouldSendToMeta(event)) {
      promises.push(this.metaCAPI.sendEvent(this.mapToMeta(event)).catch(e => this.logError('Meta', e)));
    }
    
    // TikTok EAPI
    if (this.shouldSendToTikTok(event)) {
      promises.push(this.tiktokEAPI.sendEvent(this.mapToTikTok(event)).catch(e => this.logError('TikTok', e)));
    }
    
    // Snap CAPI
    if (this.shouldSendToSnap(event)) {
      promises.push(this.snapCAPI.sendEvent(this.mapToSnap(event)).catch(e => this.logError('Snap', e)));
    }
    
    await Promise.allSettled(promises);
    
    // تحديث حالة الإرسال في DB
    await this.updateDispatchStatus(event, 'completed');
  }

  private shouldSendToMeta(event: AnalyticsEvent): boolean {
    const metaEvents = ['purchase', 'lead', 'view_content', 'add_to_cart', 'initiate_checkout', 'complete_registration', 'contact'];
    return metaEvents.includes(this.mapEventName(event.event));
  }
}
```

### 11.3 GA4 Measurement Protocol (Server)

```typescript
// lib/analytics/ga4-server.service.ts
@Injectable()
export class GA4ServerService {
  private readonly measurementId = process.env.GA4_MEASUREMENT_ID;
  private readonly apiSecret = process.env.GA4_API_SECRET;
  private readonly apiUrl = `https://www.google-analytics.com/mp/collect?measurement_id=${this.measurementId}&api_secret=${this.apiSecret}`;

  async sendEvent(event: AnalyticsEvent): Promise<void> {
    const payload = {
      client_id: event.server_cookie_fpc || event.session_id,
      user_id: event.user_id,
      timestamp_micros: Date.now() * 1000,
      non_personalized_ads: !event.consent_status?.advertising,
      
      user_properties: {
        user_type: { value: event.user_type },
        city: { value: event.city },
      },
      
      events: [{
        name: this.sanitizeEventName(event.event),
        params: {
          ...event,
          currency: 'SAR',
          // إزالة الحقول غير المسموحة
        },
      }],
    };

    await fetch(this.apiUrl, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(payload),
    });
  }

  private sanitizeEventName(name: string): string {
    // GA4 يسمح فقط بـ [a-zA-Z0-9_] وطول 40
    return name.replace(/[^a-zA-Z0-9_]/g, '_').substring(0, 40);
  }
}
```

```typescript
export function CookieConsent() {

    // إظهار البانر إذا لم يوافق المستخدم بعد

      necessary: true,
      analytics: true,
      advertising: true,
      functional: true,
    };
    saveConsent(consent);
    updateConsentMode(consent);
    setShowBanner(false);

  const handleRejectAll = () => {
    const consent: ConsentState = {
      necessary: true,  // دائماً true
      analytics: false,
      advertising: false,
      functional: true,
    };
    saveConsent(consent);
    updateConsentMode(consent);

    analytics.track({ event: 'consent_given', consent_type: 'necessary_only' });
  };

  const handleSavePreferences = (prefs: ConsentState) => {
    saveConsent(prefs);
    updateConsentMode(prefs);
    setShowBanner(false);
  };

  if (!showBanner) return null;

  return (
```
