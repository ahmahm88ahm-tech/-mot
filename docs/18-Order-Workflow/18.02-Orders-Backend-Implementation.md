```typescript
// apps/api/src/modules/orders/dto/create-order.dto.ts
import { z } from 'zod';
import { createZodDto } from 'nestjs-zod';

/**
 * Coordinates schema — matches 21-Maps §4 precision requirements
 */
export const CoordinatesSchema = z.object({
  latitude: z.number({ required_error: 'خطأ: خط العرض مطلوب' })
    .min(-90, 'خطأ: خط العرض يجب أن يكون بين -90 و 90')
    .max(90, 'خطأ: خط العرض يجب أن يكون بين -90 و 90'),
  longitude: z.number({ required_error: 'خطأ: خط الطول مطلوب' })
    .min(-180, 'خطأ: خط الطول يجب أن يكون بين -180 و 180')
    .max(180, 'خطأ: خط الطول يجب أن يكون بين -180 و 180'),
});

/**
 * Address input — matches 06-API §7.1 CreateOrder request body
 */
export const AddressInputSchema = z.object({
  address: z.string({ required_error: 'خطأ: العنوان مطلوب' })
    .min(5, 'خطأ: العنوان قصير جداً (5 أحرف على الأقل)')
    .max(500, 'خطأ: العنوان طويل جداً (500 حرف كحد أقصى)')
    .trim(),
  coordinates: CoordinatesSchema,
  cityId: z.string().cuid('خطأ: معرف المدينة غير صالح').optional(),
  districtId: z.string().cuid('خطأ: معرف الحي غير صالح').optional(),
  floor: z.number().int().min(0, 'خطأ: الطابق لا يمكن أن يكون سالباً').max(200).optional(),
  hasElevator: z.boolean().default(false),
  notes: z.string().max(1000, 'خطأ: الملاحظات طويلة جداً (1000 حرف كحد أقصى)').trim().optional(),
});

/**
 * CreateOrder schema — strict mode prevents mass assignment (09-Security §8.1)
 */
export const CreateOrderSchema = z.object({
  serviceId: z.string({ required_error: 'خطأ: الخدمة مطلوبة' })
    .cuid('خطأ: معرف الخدمة غير صالح'),

  fromAddress: AddressInputSchema.refine(
    (addr) => addr.coordinates.latitude !== 0 && addr.coordinates.longitude !== 0,
    { message: 'خطأ: إحداثيات موقع الاستلام غير صالحة' },
  ),

  toAddress: AddressInputSchema.refine(
    (addr) => addr.coordinates.latitude !== 0 && addr.coordinates.longitude !== 0,
    { message: 'خطأ: إحداثيات موقع التسليم غير صالحة' },
  ),

  scheduledDate: z.string({ required_error: 'خطأ: تاريخ الموعد مطلوب' })
    .datetime({ message: 'خطأ: صيغة التاريخ غير صالحة (ISO 8601)' })
    .refine(
      (val) => new Date(val) > new Date(Date.now() + 2 * 60 * 60 * 1000), // at least 2 hours ahead
      { message: 'خطأ: الموعد يجب أن يكون بعد ساعتين على الأقل من الآن' },
    ),

  scheduledSlot: z.enum(['morning', 'afternoon', 'evening'], {
    errorMap: () => ({ message: 'خطأ: الفترة يجب أن تكون صباحاً أو ظهراً أو مساءً' }),
  }).optional(),

  priority: z.enum(['LOW', 'NORMAL', 'HIGH', 'URGENT']).default('NORMAL'),

  customerNotes: z.string()
    .max(2000, 'خطأ: الملاحظات طويلة جداً (2000 حرف كحد أقصى)')
    .trim()
    .optional(),

  teamSize: z.number().int()
    .min(1, 'خطأ: عدد العمال يجب أن يكون 1 على الأقل')
    .max(10, 'خطأ: عدد العمال يجب أن يكون 10 كحد أقصى')
    .default(2),

  vehicleType: z.enum(['small', 'medium', 'large', 'crane']).optional(),
}).strict(); // ❌ يمنع أي حقول إضافية (mass assignment protection)

// Cross-field validation: from ≠ to
export const CreateOrderSchemaWithCrossValidation = CreateOrderSchema.refine(
  (data) => {
    const fromLat = data.fromAddress.coordinates.latitude;
    const fromLng = data.fromAddress.coordinates.longitude;
    const toLat = data.toAddress.coordinates.latitude;
    const toLng = data.toAddress.coordinates.longitude;
    // Ensure not same location (within ~100m)
    const dist = Math.sqrt(Math.pow(fromLat - toLat, 2) + Math.pow(fromLng - toLng, 2));
    return dist > 0.001; // ~100m minimum
  },
  { message: 'خطأ: موقع الاستلام والتسليم يجب أن يكونا مختلفين', path: ['toAddress'] },
);

export class CreateOrderDto extends createZodDto(CreateOrderSchemaWithCrossValidation) {}
export type CreateOrderInput = z.infer<typeof CreateOrderSchemaWithCrossValidation>;
```

## 2. Order State Machine

### 📁 الملف 2.1: `apps/api/src/modules/orders/order-state-machine.service.ts`

```typescript
// apps/api/src/modules/orders/order-state-machine.service.ts
import { Injectable, BadRequestException, Logger } from '@nestjs/common';
import { PrismaService } from '../../prisma/prisma.service';
import { EventEmitter2 } from '@nestjs/event-emitter';
import {
  VALID_TRANSITIONS,
  ADMIN_APPROVAL_CANCEL_STATUSES,
  TERMINAL_STATUSES,
  type OrderStatus,
} from './constants';

export interface TransitionResult {
  success: boolean;
  previousStatus: OrderStatus;
  newStatus: OrderStatus;
}

@Injectable()
export class OrderStateMachineService {
  private readonly logger = new Logger(OrderStateMachineService.name);

  constructor(
    private readonly prisma: PrismaService,
    private readonly eventEmitter: EventEmitter2,
  ) {}

  /**
   * Validate and execute a state transition atomically.
   * Returns the transition result or throws BadRequestException.
   */
  async transition(
    orderId: string,
    targetStatus: OrderStatus,
    changedBy: string,
    changedByRole: string,
    reason?: string,
    location?: { latitude: number; longitude: number },
  ): Promise<TransitionResult> {
    // 1. Fetch current order status
    const order = await this.prisma.order.findUnique({
      where: { id: orderId },
      select: { id: true, status: true, orderNumber: true, customerId: true },
    });

    if (!order) {
      throw new BadRequestException('الطلب غير موجود');
    }

    const currentStatus = order.status as OrderStatus;

    // 2. Check if already in terminal state
    if (TERMINAL_STATUSES.includes(currentStatus)) {
      throw new BadRequestException(
        `لا يمكن تغيير حالة الطلب "${order.orderNumber}" لأنه ${currentStatus === 'COMPLETED' ? 'مكتمل' : 'ملغي'} بالفعل`,
      );
    }

    // 3. Validate transition
    const allowedNext = VALID_TRANSITIONS[currentStatus];
    if (!allowedNext || !allowedNext.includes(targetStatus)) {
      throw new BadRequestException(
        `لا يمكن الانتقال من "${this.statusLabel(currentStatus)}" إلى "${this.statusLabel(targetStatus)}". ` +
        `الحالات المسموحة: ${allowedNext.map(s => this.statusLabel(s)).join(', ')}`,
      );
    }

    // 4. Check admin approval for late cancellations
    if (targetStatus === 'CANCELLED' && ADMIN_APPROVAL_CANCEL_STATUSES.includes(currentStatus)) {
      if (!['ADMIN', 'MANAGER'].includes(changedByRole)) {
        throw new BadRequestException(
          'إلغاء الطلب بعد تعيين السائق يتطلب موافقة المشرف. تواصل مع الدعم.',
        );
      }
    }

    // 5. Execute transition in a transaction
    const result = await this.prisma.$transaction(async (tx) => {
      // Update order status
      const updated = await tx.order.update({
        where: { id: orderId },
        data: {
          status: targetStatus,
          ...(targetStatus === 'COMPLETED' ? { completedAt: new Date() } : {}),
          ...(targetStatus === 'CANCELLED' ? {
            cancelledAt: new Date(),
            cancellationReason: reason || 'تم الإلغاء',
            cancelledBy: changedBy,
          } : {}),
        },
      });

      // Log status history
      await tx.orderStatusHistory.create({
        data: {
          orderId,
          fromStatus: currentStatus,
          toStatus: targetStatus,
          changedBy,
          changedByRole,
          reason: reason || null,
          latitude: location?.latitude ?? null,
          longitude: location?.longitude ?? null,
        },
      });

      return {
        success: true,
        previousStatus: currentStatus,
        newStatus: targetStatus,
        orderNumber: order.orderNumber,
        customerId: order.customerId,
      };
    });

    // 6. Emit domain event (outside transaction to avoid blocking)
    this.emitStatusEvent(result);

    this.logger.log(
      `Order ${order.orderNumber}: ${currentStatus} → ${targetStatus} by ${changedByRole}:${changedBy}`,
    );

    return result;
  }

  /**
   * Get allowed next states for a given status
   */
  getAllowedTransitions(status: OrderStatus): OrderStatus[] {
    return VALID_TRANSITIONS[status] || [];
  }

  /**
   * Check if a transition is valid without executing it
   */
  canTransition(from: OrderStatus, to: OrderStatus): boolean {
    return VALID_TRANSITIONS[from]?.includes(to) ?? false;
  }

  private emitStatusEvent(result: any) {
    const eventName = `order.${result.newStatus.toLowerCase()}`;
    this.eventEmitter.emit(eventName, {
      orderId: result.orderNumber, // Use orderNumber for external events
      userId: result.customerId,
      orderNumber: result.orderNumber,
      previousStatus: result.previousStatus,
      newStatus: result.newStatus,
    });
  }

  private statusLabel(status: OrderStatus): string {
    const labels: Record<OrderStatus, string> = {
      CREATED: 'تم الإنشاء', CONFIRMED: 'تم التأكيد', ASSIGNED: 'تم التعيين',
      DRIVER_STARTED: 'السائق في الطريق', ARRIVED: 'وصل السائق',
      LOADING: 'جاري التحميل', MOVING: 'في الطريق', UNLOADING: 'جاري التفريغ',
      INSTALLATION: 'جاري التركيب', COMPLETED: 'مكتمل', CANCELLED: 'ملغي',
    };
    return labels[status] || status;
  }
}
```

## 3. Orders Repository

### 📁 الملف 3.1: `apps/api/src/modules/orders/orders.repository.ts`

```typescript
// apps/api/src/modules/orders/orders.repository.ts
import { Injectable, Logger } from '@nestjs/common';
import { PrismaService } from '../../prisma/prisma.service';
import type { CreateOrderInput } from './dto/create-order.dto';
import type { OrderPricing } from './types';

@Injectable()
export class OrdersRepository {
  private readonly logger = new Logger(OrdersRepository.name);

  constructor(private readonly prisma: PrismaService) {}

  /**
   * Generate unique order number: MTQ-{YEAR}-{SEQUENCE}
   */
  async generateOrderNumber(): Promise<string> {
    const year = new Date().getFullYear();
    const startOfYear = new Date(`${year}-01-01T00:00:00Z`);

    const count = await this.prisma.order.count({
      where: { createdAt: { gte: startOfYear } },
    });

    return `MTQ-${year}-${String(count + 1).padStart(5, '0')}`;
  }

  /**
   * Create order with all related data in a single transaction
   */
  async create(data: {
    input: CreateOrderInput;
    customerId: string;
    orderNumber: string;
    pricing: OrderPricing;
    distanceKm: number;
    durationMin: number;
  }) {
    const { input, customerId, orderNumber, pricing, distanceKm, durationMin } = data;

    return this.prisma.order.create({
      data: {
        orderNumber,
        customerId,
        serviceId: input.serviceId,

        // From
        fromAddress: input.fromAddress.address,
        fromLatitude: input.fromAddress.coordinates.latitude,
        fromLongitude: input.fromAddress.coordinates.longitude,
        fromFloor: input.fromAddress.floor ?? null,
        fromHasElevator: input.fromAddress.hasElevator ?? false,
        fromNotes: input.fromAddress.notes ?? null,
        fromCityId: input.fromAddress.cityId ?? null,
        fromDistrictId: input.fromAddress.districtId ?? null,

        // To
        toAddress: input.toAddress.address,
        toLatitude: input.toAddress.coordinates.latitude,
        toLongitude: input.toAddress.coordinates.longitude,
        toFloor: input.toAddress.floor ?? null,
        toHasElevator: input.toAddress.hasElevator ?? false,
        toNotes: input.toAddress.notes ?? null,
        toCityId: input.toAddress.cityId ?? null,
        toDistrictId: input.toAddress.districtId ?? null,

        // Route
        distanceKm,
        durationMin,

        // Pricing
        basePrice: pricing.basePrice,
        distancePrice: pricing.distancePrice,
        extraServicesPrice: pricing.extraServicesPrice,
        discountAmount: pricing.discountAmount,
        taxAmount: pricing.taxAmount,
        totalAmount: pricing.totalAmount,
        currency: pricing.currency,

        // Schedule
        scheduledDate: new Date(input.scheduledDate),
        scheduledSlot: input.scheduledSlot ?? null,

        // Execution
        teamSize: input.teamSize ?? 2,
        vehicleType: input.vehicleType ?? null,

        // Notes
        customerNotes: input.customerNotes ?? null,

        // Metadata
        priority: input.priority ?? 'NORMAL',
      },
      include: {
        service: { select: { id: true, nameAr: true, slug: true, thumbnailUrl: true } },
        fromCity: { select: { id: true, nameAr: true, slug: true } },
        toCity: { select: { id: true, nameAr: true, slug: true } },
      },
    });
  }

  /**
   * Find by ID with full details (admin view)
   */
  async findByIdFull(id: string) {
    return this.prisma.order.findUnique({
      where: { id, deletedAt: null },
      include: {
        service: true,
        customer: { select: { id: true, firstName: true, lastName: true, phone: true, email: true } },
        assignedTo: { select: { id: true, firstName: true, lastName: true, phone: true, avatarUrl: true } },
        fromCity: { select: { id: true, nameAr: true } },
        toCity: { select: { id: true, nameAr: true } },
        images: { orderBy: { sortOrder: 'asc' } },
        statusHistory: { orderBy: { createdAt: 'desc' } },
      },
    });
  }

  /**
   * Find by ID with customer-safe details (no internal notes)
   */
  async findByIdForCustomer(id: string, customerId: string) {
    return this.prisma.order.findFirst({
      where: { id, customerId, deletedAt: null },
      include: {
        service: { select: { id: true, nameAr: true, slug: true, thumbnailUrl: true } },
        fromCity: { select: { id: true, nameAr: true } },
        toCity: { select: { id: true, nameAr: true } },
        assignedTo: { select: { id: true, firstName: true, lastName: true, phone: true, avatarUrl: true } },
        images: { where: { type: { not: 'damage' } }, orderBy: { sortOrder: 'asc' } }, // hide damage photos from customer
        statusHistory: { orderBy: { createdAt: 'desc' }, take: 10 },
      },
    });
  }

  /**
   * Paginated list for customer
   */
  async findByCustomer(customerId: string, options: {
    page: number; limit: number; status?: string;
  }) {
    const where: any = { customerId, deletedAt: null };
    if (options.status) where.status = options.status;

    const [data, total] = await Promise.all([
      this.prisma.order.findMany({
        where,
        orderBy: { createdAt: 'desc' },
        skip: (options.page - 1) * options.limit,
        take: options.limit,
        select: {
          id: true, orderNumber: true, status: true, totalAmount: true, currency: true,
          scheduledDate: true, createdAt: true, rating: true,
          service: { select: { nameAr: true, thumbnailUrl: true } },
          fromCity: { select: { nameAr: true } },
          toCity: { select: { nameAr: true } },
        },
      }),
      this.prisma.order.count({ where }),
    ]);

    return {
      data,
      pagination: {
        page: options.page,
        limit: options.limit,
        total,
        totalPages: Math.ceil(total / options.limit),
        hasNext: options.page * options.limit < total,
        hasPrev: options.page > 1,
      },
    };
  }

  /**
   * Paginated list for admin with filters
   */
  async findAllAdmin(options: {
    page: number; limit: number; status?: string;
    search?: string; cityId?: string; branchId?: string;
    dateFrom?: string; dateTo?: string;
  }) {
    const where: any = { deletedAt: null };

    if (options.status) where.status = options.status;
    if (options.cityId) where.OR = [{ fromCityId: options.cityId }, { toCityId: options.cityId }];
    if (options.branchId) where.branchId = options.branchId;
    if (options.dateFrom || options.dateTo) {
      where.createdAt = {};
      if (options.dateFrom) where.createdAt.gte = new Date(options.dateFrom);
      if (options.dateTo) where.createdAt.lte = new Date(options.dateTo);
    }
    if (options.search) {
      where.OR = [
        { orderNumber: { contains: options.search, mode: 'insensitive' } },
        { customer: { firstName: { contains: options.search, mode: 'insensitive' } } },
        { customer: { lastName: { contains: options.search, mode: 'insensitive' } } },
        { customer: { phone: { contains: options.search } } },
      ];
    }

    const [data, total] = await Promise.all([
      this.prisma.order.findMany({
        where,
        orderBy: { createdAt: 'desc' },
        skip: (options.page - 1) * options.limit,
        take: options.limit,
        select: {
          id: true, orderNumber: true, status: true, priority: true,
          totalAmount: true, currency: true, scheduledDate: true, createdAt: true,
          distanceKm: true,
          customer: { select: { id: true, firstName: true, lastName: true, phone: true } },
          assignedTo: { select: { id: true, firstName: true, lastName: true } },
          service: { select: { nameAr: true } },
          fromCity: { select: { nameAr: true } },
          toCity: { select: { nameAr: true } },
        },
      }),
      this.prisma.order.count({ where }),
    ]);

    return {
      data,
      pagination: {
        page: options.page, limit: options.limit, total,
        totalPages: Math.ceil(total / options.limit),
      },
    };
  }

  /**
   * Dashboard statistics
   */
  async getDashboardStats() {
    const now = new Date();
    const startOfMonth = new Date(now.getFullYear(), now.getMonth(), 1);
    const startOfDay = new Date(now.getFullYear(), now.getMonth(), now.getDate());

    const [
      totalOrders, activeOrders, completedOrders, todayOrders,
      revenueResult, monthlyRevenueResult, avgRatingResult,
      newCustomersThisMonth, recentOrders,
    ] = await Promise.all([
      this.prisma.order.count({ where: { deletedAt: null } }),
      this.prisma.order.count({ where: { deletedAt: null, status: { notIn: ['COMPLETED', 'CANCELLED'] } } }),
      this.prisma.order.count({ where: { deletedAt: null, status: 'COMPLETED' } }),
      this.prisma.order.count({ where: { deletedAt: null, createdAt: { gte: startOfDay } } }),
      this.prisma.order.aggregate({ where: { deletedAt: null, status: 'COMPLETED' }, _sum: { totalAmount: true } }),
      this.prisma.order.aggregate({ where: { deletedAt: null, status: 'COMPLETED', completedAt: { gte: startOfMonth } }, _sum: { totalAmount: true } }),
      this.prisma.order.aggregate({ where: { deletedAt: null, rating: { not: null } }, _avg: { rating: true } }),
      this.prisma.user.count({ where: { role: 'CUSTOMER', createdAt: { gte: startOfMonth } } }),
      this.prisma.order.findMany({
        where: { deletedAt: null },
        orderBy: { createdAt: 'desc' },
        take: 10,
        select: {
          id: true, orderNumber: true, status: true, totalAmount: true, createdAt: true,
          customer: { select: { firstName: true, lastName: true } },
        },
      }),
    ]);

    return {
      totalOrders,
      activeOrders,
      completedOrders,
      todayOrders,
      totalRevenue: revenueResult._sum.totalAmount ?? 0,
      monthlyRevenue: monthlyRevenueResult._sum.totalAmount ?? 0,
      avgRating: avgRatingResult._avg.rating ?? 0,
      newCustomersThisMonth,
      recentOrders: recentOrders.map(o => ({
        ...o,
        customerName: `${o.customer.firstName} ${o.customer.lastName}`,
      })),
    };
  }
}
```

```typescript
// apps/api/src/modules/orders/orders.service.ts
import { Injectable, NotFoundException, ForbiddenException, BadRequestException, Logger } from '@nestjs/common';
import { OrdersRepository } from './orders.repository';
import { OrderStateMachineService } from './order-state-machine.service';
import { CacheService } from '../../common/cache/cache.service';
import type { CreateOrderInput } from './dto/create-order.dto';
import type { CalculatePriceDto } from './dto/calculate-price.dto';
import type { AssignDriverDto } from './dto/assign-driver.dto';
import type { OrderPricing } from './types';
import { PRICE_PER_KM, TAX_RATE, MIN_ORDER_AMOUNT } from './constants';

@Injectable()
export class OrdersService {
  private readonly logger = new Logger(OrdersService.name);

  constructor(
    private readonly repo: OrdersRepository,
    private readonly stateMachine: OrderStateMachineService,
    private readonly cache: CacheService,
  ) {}

  // ===== CREATE =====

  async create(input: CreateOrderInput, customerId: string) {
    // 1. Validate service exists and is active
    const service = await this.cache.wrap(
      CacheService.serviceKey(input.serviceId),
      300,
      async () => {
        const s = await this.repo['prisma'].service.findUnique({ where: { id: input.serviceId } });
        if (!s || !s.isActive || s.deletedAt) {
          throw new BadRequestException('الخدمة المطلوبة غير متوفرة حالياً');
        }
        return s;
      },
    );

    // 2. Calculate distance (Haversine)
    const distanceKm = this.haversineDistance(
      input.fromAddress.coordinates,
      input.toAddress.coordinates,
    );

    // 3. Calculate pricing
    const pricing = this.computePricing({
      basePrice: Number(service.basePrice),
      distanceKm,
      taxRate: Number(service.taxRate || TAX_RATE),
      floorSurcharge: this.calculateFloorSurcharge(input.fromAddress.floor, input.toAddress.floor, input.fromAddress.hasElevator, input.toAddress.hasElevator),
    });

    // 4. Generate order number
    const orderNumber = await this.repo.generateOrderNumber();

    // 5. Create order
    const order = await this.repo.create({
      input,
      customerId,
      orderNumber,
      pricing,
      distanceKm,
      durationMin: Math.max(30, Math.round(distanceKm * 2.5)), // min 30 min
    });

    // 6. Log initial status
    await this.stateMachine.transition(order.id, 'CREATED', customerId, 'CUSTOMER', 'تم إنشاء الطلب عبر الموقع');

    // 7. Invalidate caches
    await this.cache.invalidateTag('orders');

    this.logger.log(`Order created: ${orderNumber} by customer:${customerId}, total: ${pricing.totalAmount} SAR`);

    return order;
  }

  // ===== FIND BY ID =====

  async findById(id: string, user: { id: string; role: string }) {
    let order;

    if (['ADMIN', 'MANAGER', 'SUPER_ADMIN'].includes(user.role)) {
      order = await this.repo.findByIdFull(id);
    } else {
      order = await this.repo.findByIdForCustomer(id, user.id);
    }

    if (!order) throw new NotFoundException('الطلب غير موجود');

    // Ownership check for non-admin users
    if (!['ADMIN', 'MANAGER', 'SUPER_ADMIN'].includes(user.role)) {
      if (order.customerId !== user.id) {
        // Also allow assigned driver
        if (order.assignedToId !== user.id) {
          throw new ForbiddenException('ليس لديك صلاحية الوصول لهذا الطلب');
        }
      }
    }

    // Add allowed transitions for UI
    const allowedTransitions = this.stateMachine.getAllowedTransitions(order.status as any);

    return { ...order, allowedTransitions };
  }

  // ===== FIND BY CUSTOMER =====

  async findByCustomer(customerId: string, options: { page: number; limit: number; status?: string }) {
    return this.repo.findByCustomer(customerId, options);
  }

  // ===== FIND ALL ADMIN =====

  async findAllAdmin(options: {
    page: number; limit: number; status?: string;
    search?: string; cityId?: string; branchId?: string;
    dateFrom?: string; dateTo?: string;
  }) {
    return this.repo.findAllAdmin(options);
  }

  // ===== UPDATE STATUS =====

  async updateStatus(orderId: string, targetStatus: string, userId: string, userRole: string, reason?: string, location?: { latitude: number; longitude: number }) {
    const result = await this.stateMachine.transition(
      orderId, targetStatus as any, userId, userRole, reason, location,
    );

    // Invalidate caches
    await this.cache.invalidateTag('orders');
    await this.cache.del(CacheService.dashboardStatsKey());

    return result;
  }

  // ===== ASSIGN DRIVER =====

  async assignDriver(orderId: string, dto: AssignDriverDto, adminId: string) {
    // Verify driver exists and is active
    const driver = await this.repo['prisma'].user.findFirst({
      where: { id: dto.driverId, role: { in: ['DRIVER', 'EMPLOYEE'] }, status: 'ACTIVE' },
    });
    if (!driver) throw new BadRequestException('السائق غير موجود أو غير نشط');

    // Update order
    const order = await this.repo['prisma'].order.update({
      where: { id: orderId },
      data: {
        assignedToId: dto.driverId,
        teamSize: dto.teamSize ?? undefined,
        vehicleType: dto.vehicleType ?? undefined,
        internalNotes: dto.internalNotes ?? undefined,
      },
    });

    // Transition to ASSIGNED
    await this.stateMachine.transition(orderId, 'ASSIGNED', adminId, 'ADMIN', `تعيين السائق: ${driver.firstName} ${driver.lastName}`);

    await this.cache.invalidateTag('orders');

    this.logger.log(`Order ${order.orderNumber}: assigned to driver:${dto.driverId} by admin:${adminId}`);

    return order;
  }

  // ===== CALCULATE PRICE =====

  async calculatePrice(dto: CalculatePriceDto): Promise<OrderPricing> {
    const service = await this.cache.wrap(
      CacheService.serviceKey(dto.serviceId),
      300,
      async () => {
        const s = await this.repo['prisma'].service.findUnique({ where: { id: dto.serviceId } });
        if (!s) throw new NotFoundException('الخدمة غير موجودة');
        return s;
      },
    );

    const distanceKm = this.haversineDistance(
      { latitude: dto.fromLatitude, longitude: dto.fromLongitude },
      { latitude: dto.toLatitude, longitude: dto.toLongitude },
    );

    return this.computePricing({
      basePrice: Number(service.basePrice),
      distanceKm,
      taxRate: Number(service.taxRate || TAX_RATE),
      floorSurcharge: this.calculateFloorSurcharge(dto.floor, dto.floor, dto.hasElevator, dto.hasElevator),
    });
  }

  // ===== DASHBOARD STATS =====

  async getDashboardStats() {
    return this.cache.wrap(CacheService.dashboardStatsKey(), 60, () => this.repo.getDashboardStats());
  }

  // ===== PRIVATE HELPERS =====

  private computePricing(params: {
    basePrice: number; distanceKm: number; taxRate: number; floorSurcharge: number;
  }): OrderPricing {
    const { basePrice, distanceKm, taxRate, floorSurcharge } = params;

    const distancePrice = Math.round(distanceKm * PRICE_PER_KM * 100) / 100;
    const extraServicesPrice = floorSurcharge;
    const subtotal = basePrice + distancePrice + extraServicesPrice;
    const taxAmount = Math.round(subtotal * taxRate * 100) / 100;
    const totalAmount = Math.max(MIN_ORDER_AMOUNT, Math.round((subtotal + taxAmount) * 100) / 100);

    return {
      basePrice,
      distancePrice,
      extraServicesPrice,
      discountAmount: 0,
      taxAmount,
      totalAmount,
      currency: 'SAR',
    };
  }

  private calculateFloorSurcharge(fromFloor?: number | null, toFloor?: number | null, fromElevator?: boolean, toElevator?: boolean): number {
    let surcharge = 0;
    const floorsWithoutElevator = (fromFloor && !fromElevator ? fromFloor : 0) + (toFloor && !toElevator ? toFloor : 0);
    if (floorsWithoutElevator > 0) {
      surcharge = floorsWithoutElevator * 25; // 25 SAR per floor without elevator
    }
    return surcharge;
  }

  private haversineDistance(a: { latitude: number; longitude: number }, b: { latitude: number; longitude: number }): number {
    const R = 6371;
    const toRad = (deg: number) => (deg * Math.PI) / 180;
    const dLat = toRad(b.latitude - a.latitude);
    const dLng = toRad(b.longitude - a.longitude);
    const x = Math.sin(dLat / 2) ** 2 + Math.cos(toRad(a.latitude)) * Math.cos(toRad(b.latitude)) * Math.sin(dLng / 2) ** 2;
    return Math.round(R * 2 * Math.atan2(Math.sqrt(x), Math.sqrt(1 - x)) * 100) / 100;
  }
}
```
