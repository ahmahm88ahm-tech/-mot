```bash
until mc alias set local "${MINIO_ENDPOINT}" "${MINIO_USER}" "${MINIO_PASSWORD}" >/dev/null 2>&1; do

if mc ls "local/${MINIO_BUCKET}" >/dev/null 2>&1; then
```

```bash
#!/bin/sh
# scripts/wait-for.sh — ينتظر صحة عنوان حتى يصبح جاهزاً
# الاستخدام: ./scripts/wait-for.sh <host:port> <label> [timeout]
HOST="$1"; LABEL="$2"; TIMEOUT="${3:-90}"; elapsed=0
printf "   ⏳ %-14s " "$LABEL"
while ! nc -z "${HOST%:*}" "${HOST#*:}" >/dev/null 2>&1; do
  sleep 1; elapsed=$((elapsed+1))
  if [ "$elapsed" -ge "$TIMEOUT" ]; then printf "❌ timeout\n"; exit 1; fi
done
printf "✅ جاهز\n"
```

```bash
#!/bin/bash
# scripts/status.sh — يفحص كل نقطة ويطبع جدول حالة (نصيّ)
BASE="http://localhost"
declare -a ROWS=(
  "postgres|قاعدة البيانات|${BASE}:5432|tcp"
  "redis|الذاكرة المؤقتة|${BASE}:6379|tcp"
  "minio|تخزين الكائنات|${BASE}:9000|tcp"
  "mailhog|البريد التطويري|${BASE}:8025|http"
  "api|الـ Backend|${BASE}:4000/health|http"
  "web|موقع العملاء|${BASE}:3000|http"
  "admin|لوحة الإدارة|${BASE}:3001|http"
)
G=$'\033[0;32m'; R=$'\033[0;31m'; Y=$'\033[0;33m'; N=$'\033[0m'; B=$'\033[1m'
printf "\n${B}  الخدمة            الحالة    المنفذ${N}\n"
printf "  ─────────────────────────────────────────\n"
for row in "${ROWS[@]}"; do
  IFS='|' read -r key label url kind <<< "$row"
  if [ "$kind" = "tcp" ]; then
    host="${url#*://}"; host="${host%:*}"; port="${url##*:}"
    nc -z "$host" "$port" >/dev/null 2>&1 && st="${G}● يعمل${N}" || st="${R}● متوقف${N}"
  else
    code=$(curl -s -o /dev/null -m 3 -w "%{http_code}" "$url" 2>/dev/null || echo 000)
    [ "$code" = "200" ] || [ "$code" = "302" ] && st="${G}● يعمل${N} " || st="${R}● ${code}${N} "
  fi
  port="${url##*:}"; port="${port%%/*}"
  printf "  %-18s %b   %s\n" "$label" "$st" "$port"
done
printf "\n"
```

````markdown
git clone <repo> && cd motanaqil
cp .env.example .env
make up            # يُقلع كل شيء + يهيّئ DB + يزرع البيانات
make console       # يفتح لوحة التشغيل الحية
```

افتح:
- العملاء → <http://localhost:3000>
- الإدارة → <http://localhost:3001>
- الـ API → <http://localhost:4000/api/v1> · Swagger → `/docs`
- البريد → <http://localhost:8025> · التخزين → <http://localhost:9001>

**دخول تجريبي** (كلمة المرور `Motanaqil@2026`):
`admin@motanaqil.com` · `customer@motanaqil.com` · `driver@motanaqil.com`

## أوامر يومية

| الأمر | الفعل |
|-------|-------|
| `make up` | إقلاع كامل مع التهيئة والزراعة |
| `make dev` | البنية التحتية فقط + تطبيقات محلية بـ `pnpm dev` |
| `make logs` | متابعة السجلات |
| `make seed` | إعادة زراعة البيانات |
| `make verify` | أطقم التحقق الثلاثة (DB · إشعارات · عقد Front↔Back) |
| `make status` | فحص حالة كل خدمة |
| `make reset` | حذف كل البيانات والإقلاع من الصفر |
| `make help` | كل الأوامر |

## ماذا يعمل فعلاً (مثبت بأطقم التحقق)

- **مصادقة كاملة** — تسجيل/دخول/MFA/تحديث/استعادة، مربوط بـ DB، بكلمات مرور bcrypt حقيقية.
- **نظام طلبات** — إنشاء/قائمة/تفاصيل/تعيين/تسعير/لوحة إحصاءات، مع State Machine من 11 حالة.
- **إشعارات 5 قنوات** — in-app (DB + WebSocket فوري) + Email/SMS/WhatsApp/Push عبر طوابير مع سجلات تسليم.
- **CMS** — صفحات/FAQ/وسائط/إعدادات، مع تخزين صور حقيقي (MinIO محلياً / S3 إنتاجاً).
- **تتبع حي** — صفحة تتبع عامة + WebSocket لموقع السائق وتحديث الحالة.
- **أمان** — JWT RS-ready + RBAC + Rate-Limiting + CSP/HSTS + تنقية مدخلات + أخطاء عربية.

التحقق:
```bash
make verify   # يشغّل: verify-api.sh · verify-notifications.sh · verify-frontend-contract.sh
```

## استكشاف الأخطاء السريع

| العَرَض | الحل |
|---------|------|
| `api` لا يقلع | `docker compose logs api` — غالباً migration؛ نفّذ `make migrate` |
| 401 على كل طلب | تأكد أن الـ Frontend يمرر Bearer (موحّد في `api-client.ts`) |
| الصور لا تُرفع | تحقّق من `minio-init` أنشأ الـ bucket: `docker compose logs minio-init` |
| الخريطة رمادية | ضع `NEXT_PUBLIC_GOOGLE_MAPS_KEY` في `.env` ثم أعد `docker compose up web` |
| منفذ مشغول | غيّر المنفذ في `.env` (مثلاً `WEB_PORT=3100`) |
| بيانات قديمة/تالفة | `make reset` |

## من التطوير إلى الإنتاج

صور الإنتاج جاهزة (`apps/*/Dockerfile`، multi-stage، non-root، healthcheck). للإنتاج:
1. استبدل كل قيم `dev-*` في `.env` بأسرار حقيقية (عبر Secrets Manager).
2. فعّل RS256 للـ JWT، وKMS لتشفير MFA.
3. `make build` ثم نشر عبر `docker-compose.prod.yml` أو ECS/K8s (راجع `11-Deployment-Guide`).
4. نفّذ `docs/PRODUCTION-CHECKLIST.md` قبل الإطلاق.
````

- **تخطيط Bento غير متماثل** بدل شبكة بطاقات متساوية أو broadsheet.
- **خلفية طبقاتية:** شبكة مُقنّعة + ضوضاء + توهّجان كهرمانيّان زاويّان — لا aurora blob.

- **تردّ على الإدخال:** أزرار نسخ بتغذية راجعة («تم ✓»)، تبويبات تتبدّل بحركة، فحص حقيقي لـ `/health` مع رجوع صادق للمحاكاة إن تعذّر.
