```
                    │  CONFIRMED  │

                           │
                           │ arrive()

                           ↓
                    ┌─────────────┐
                    │   ARRIVED   │
                    └──────┬──────┘
                           │
                           │ start_loading()
                           │ [employee_confirms]
                           ↓
                    ┌─────────────┐
                    │   LOADING   │
                    └──────┬──────┘
                           │
                           │ start_moving()
                           │ [loading_complete + photos_uploaded]
                           ↓
                    ┌─────────────┐

                    └──────┬──────┘
                           │
                           │ start_unloading()

         ┌─────────────────────────────────────────┐
         │                                         │
         │         cancel(reason)                  │
         │         [any_state except COMPLETED]    │
         │                                         │
```

## 2. State Definitions

```typescript
interface CreatedState {
  status: 'created';
  allowedTransitions: ['confirmed', 'cancelled'];
  triggers: ['confirm', 'cancel'];
  guards: {
    confirm: 'payment_received OR admin_confirm';
    cancel: 'always_allowed';
  };
  actions: {
    onEnter: ['send_confirmation_email', 'create_invoice'];
    onExit: ['update_order_status'];
  };
}
```

## 3. Transition Rules Matrix

| من حالة | إلى حالة | الحدث | الشروط | الإجراء |
|--------|---------|------|--------|---------|
| CREATED | CONFIRMED | confirm | دفع عربون أو تأكيد إداري | إرسال تأكيد |
| CREATED | CANCELLED | cancel | دائماً مسموح | إلغاء الطلب |
| CONFIRMED | ASSIGNED | assign | موظف متاح ويقبل | تعيين موظف |
| CONFIRMED | CANCELLED | cancel | دائماً مسموح | إلغاء الطلب |
| ASSIGNED | DRIVER_STARTED | start_driver | الموظف يؤكد | بدء التتبع |
| ASSIGNED | CANCELLED | cancel | موافقة إدارية | إلغاء + إشعار |
| DRIVER_STARTED | ARRIVED | arrive | المسافة < 500م | إشعار وصول |
| DRIVER_STARTED | CANCELLED | cancel | موافقة إدارية | إلغاء + إشعار |
| ARRIVED | LOADING | start_loading | الموظف يؤكد | بدء التحميل |
| ARRIVED | CANCELLED | cancel | موافقة إدارية | إلغاء + إشعار |
| LOADING | MOVING | start_moving | اكتمال التحميل + صور | بدء الحركة |
| LOADING | CANCELLED | cancel | موافقة إدارية | إلغاء + إشعار |
| MOVING | UNLOADING | start_unloading | الوصول للوجهة | بدء التفريغ |
| MOVING | CANCELLED | cancel | موافقة إدارية | إلغاء + إشعار |
| UNLOADING | INSTALLATION | start_installation | الخدمة تشمل التركيب | بدء التركيب |
| UNLOADING | COMPLETED | complete | اكتمال التفريغ | إنهاء الطلب |
| UNLOADING | CANCELLED | cancel | موافقة إدارية | إلغاء + إشعار |
| INSTALLATION | COMPLETED | complete | اكتمال التركيب + تأكيد + صور | إنهاء الطلب |
| INSTALLATION | CANCELLED | cancel | موافقة إدارية | إلغاء + إشعار |

## 4. Guards (الشروط)

### 4.1 payment_received
```typescript
function guardPaymentReceived(order: Order): boolean {
  return order.payment_status === 'paid' || order.payment_status === 'partial';
}
```

### 4.2 admin_confirm
```typescript
function guardAdminConfirm(user: User): boolean {
  return user.role === 'admin' || user.role === 'manager';
}
```

### 4.3 employee_available
```typescript
async function guardEmployeeAvailable(employeeId: string, dateTime: Date): Promise<boolean> {
  const employee = await getEmployee(employeeId);
  return employee.is_active && 
         employee.availability.isOnDuty &&
         !hasConflictingOrders(employeeId, dateTime);
}
```

### 4.4 distance_from_location
```typescript
function guardDistanceFromLocation(order: Order, employeeLocation: Coordinates): boolean {
  const distance = calculateDistance(employeeLocation, order.from_coordinates);
  return distance < 0.5; // 500 meters
}
```

### 4.5 loading_complete
```typescript
async function guardLoadingComplete(order: Order): Promise<boolean> {
  const photos = await getOrderImages(order.id, 'before');
  return photos.length >= 2;
}
```

### 4.6 service_includes_installation
```typescript
async function guardServiceIncludesInstallation(order: Order): Promise<boolean> {
  const service = await getService(order.service_id);
  return service.name_ar.includes('تركيب');
}
```

### 4.7 customer_confirms

## 6. Implementation (Express/TypeScript)

```typescript
// order-state-machine.service.ts
export class OrderStateMachineService {
  private stateTransitions: Map<OrderStatus, Set<OrderStatus>> = new Map([
    ['created', new Set(['confirmed', 'cancelled'])],
    ['confirmed', new Set(['assigned', 'cancelled'])],
    ['assigned', new Set(['driver_started', 'cancelled'])],
    ['driver_started', new Set(['arrived', 'cancelled'])],
    ['arrived', new Set(['loading', 'cancelled'])],
    ['loading', new Set(['moving', 'cancelled'])],
    ['moving', new Set(['unloading', 'cancelled'])],
    ['unloading', new Set(['installation', 'completed', 'cancelled'])],
    ['installation', new Set(['completed', 'cancelled'])],
    ['completed', new Set([])],
    ['cancelled', new Set([])],
  ]);

  async transition(
    orderId: string,
    newStatus: OrderStatus,
    userId: string,
    metadata?: any
  ): Promise<Order> {
    const order = await this.ordersRepo.findById(orderId);
    const currentStatus = order.status;

    // 1. Validate transition
    if (!this.isValidTransition(currentStatus, newStatus)) {
      throw new Error(
        `Invalid transition from ${currentStatus} to ${newStatus}`
      );
    }

    // 2. Execute guards
    await this.executeGuards(order, newStatus, userId);

    // 3. Update status
    order.status = newStatus;
    await this.ordersRepo.save(order);

    // 4. Log transition
    await this.statusHistoryRepo.create({
      order_id: orderId,
      from_status: currentStatus,
      to_status: newStatus,
      changed_by: userId,
      reason: metadata?.reason,
    });

    // 5. Execute actions
    await this.executeActions(order, newStatus);

    return order;
  }

  private isValidTransition(from: OrderStatus, to: OrderStatus): boolean {
    return this.stateTransitions.get(from)?.has(to) ?? false;
  }
```

## 7. Testing

```typescript
describe('OrderStateMachineService', () => {
  it('should allow transition from created to confirmed', async () => {
    const order = await service.transition(
      orderId,
      'confirmed',
      userId,
      { reason: 'Payment received' }
    );
    expect(order.status).toBe('confirmed');
  });

  it('should reject transition from completed to cancelled', async () => {
    await expect(
      service.transition(orderId, 'cancelled', userId)
    ).rejects.toThrow();
  });

  it('should execute guards before transition', async () => {
    // Mock guard failure
    await expect(
      service.transition(orderId, 'assigned', userId)
    ).rejects.toThrow();
  });

  it('should log status history', async () => {
    await service.transition(orderId, 'confirmed', userId);
    const history = await statusHistoryRepo.findByOrder(orderId);
    expect(history.length).toBeGreaterThan(0);
  });
});
```
